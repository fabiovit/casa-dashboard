  async _loadEntityConfig(force=false) {
    if((this._entityConfigLoaded && !force) || this._entityConfigLoading) return;
    this._entityConfigLoading=true;
    try {
      let entities=null;
      let labels=null;
      let overview=null;
      if(this._hass?.callWS){
        try {
          const result=await this._hass.callWS({type:"casa_dashboard_community/config/get"});
          if(result?.entities && typeof result.entities==="object") entities=result.entities;
          if(result?.labels && typeof result.labels==="object") labels=result.labels;
          if(result?.overview && typeof result.overview==="object") overview=result.overview;
          if(Array.isArray(result?.rooms)) this._rooms=result.rooms.map(room=>({
            ...room,
            entities:Array.isArray(room?.entities)?[...room.entities]:[],
            modules:Array.isArray(room?.modules)?[...room.modules]:[],
            entity_labels:(room?.entity_labels && typeof room.entity_labels==="object")?{...room.entity_labels}:{},
            entity_images:(room?.entity_images && typeof room.entity_images==="object")?{...room.entity_images}:{},
            entity_icons:(room?.entity_icons && typeof room.entity_icons==="object")?{...room.entity_icons}:{},
            entity_sizes:(room?.entity_sizes && typeof room.entity_sizes==="object")?{...room.entity_sizes}:{},
            entity_order:Array.isArray(room?.entity_order)?[...room.entity_order]:[],
            title_size:room?.title_size||"normal",
            show_subtitle:room?.show_subtitle!==false,
            subtitle:room?.subtitle||""
          }));
        } catch(_wsErr) { /* fallback below */ }
      }
      if(!entities){
        const res=await fetch(`${ENTITY_CONFIG_URL}?v=${Date.now()}`, {cache:"no-store"});
        if(res.ok){
          const json=await res.json();
          if(json?.entities && typeof json.entities==="object") entities=json.entities;
          if(json?.labels && typeof json.labels==="object") labels=json.labels;
          if(json?.overview && typeof json.overview==="object") overview=json.overview;
          if(Array.isArray(json?.rooms)) this._rooms=json.rooms.map(room=>({
            ...room,
            entities:Array.isArray(room?.entities)?[...room.entities]:[],
            modules:Array.isArray(room?.modules)?[...room.modules]:[],
            entity_labels:(room?.entity_labels && typeof room.entity_labels==="object")?{...room.entity_labels}:{},
            entity_images:(room?.entity_images && typeof room.entity_images==="object")?{...room.entity_images}:{},
            entity_icons:(room?.entity_icons && typeof room.entity_icons==="object")?{...room.entity_icons}:{},
            entity_sizes:(room?.entity_sizes && typeof room.entity_sizes==="object")?{...room.entity_sizes}:{},
            entity_order:Array.isArray(room?.entity_order)?[...room.entity_order]:[],
            title_size:room?.title_size||"normal",
            show_subtitle:room?.show_subtitle!==false,
            subtitle:room?.subtitle||""
          }));
        }
      }
      this._entityMap=entities || {};
      this._entityLabels=labels || {};
      this._overviewConfig=overview || {};
    } catch(_err) {
      this._entityMap={};
      this._entityLabels={};
      this._overviewConfig={};
    } finally {
      this._entityConfigLoaded=true;
      this._entityConfigLoading=false;
      this._renderPage(false);
    }
  }

  _overviewDisplayName(logicalKey, entityId=null) {
    const custom=String(this._entityLabels?.[logicalKey]||"").trim();

  _configNotice() {
    const configured=this._configuredCount();
    const total=ENTITY_KEYS.size;
    const title=configured===0?"Configura le entità":"Entità configurate";
    const text=configured===0
      ? `Sono disponibili ${total} associazioni opzionali. Configura solo le funzioni che utilizzi nel tuo Home Assistant.`
      : `${configured} di ${total} associazioni disponibili configurate. Non è necessario compilarle tutte: configura solo quelle che utilizzi.`;
    const tone=configured===0?"warn":"info";
    const icon=configured===0?"mdi:cog-alert-outline":"mdi:check-circle-outline";
    return `<div class="alert ${tone} config-alert"><ha-icon icon="${icon}"></ha-icon><div><strong>${title}</strong><span>${text}</span></div><button class="config-open-btn" data-action="entity-config" type="button"><ha-icon icon="mdi:tune-variant"></ha-icon> Configura</button></div>`;
  }

  _configLabel(key) {
    const labels={
      "sensor.vehicle_battery_level":"Batteria auto",
      "sensor.vehicle_range":"Autonomia auto",
      "sensor.vehicle_charge_remaining_minutes":"Tempo ricarica restante",
      "sensor.vehicle_odometer":"Chilometraggio auto",
      "sensor.vehicle_charging_power":"Potenza ricarica auto",
      "sensor.vehicle_session_energy":"Energia sessione auto"
    };
    if(labels[key]) return labels[key];
    const pretty=key.split(".")[1]?.replaceAll("_"," ") || key;
    return pretty.replace(/\b\w/g,c=>c.toUpperCase());
  }

  _roomEntitySuggestions(query="", selected=[]){
    const q=String(query||"").trim().toLocaleLowerCase("it");
    const chosen=new Set(selected||[]);
    return Object.entries(this._hass?.states||{})
      .filter(([id])=>!chosen.has(id))
      .map(([id,obj])=>({id,name:String(obj?.attributes?.friendly_name||id)}))
      .filter(x=>!q || x.id.toLocaleLowerCase("it").includes(q) || x.name.toLocaleLowerCase("it").includes(q))
      .sort((a,b)=>{
        const an=a.name.toLocaleLowerCase("it"), bn=b.name.toLocaleLowerCase("it");
        const ae=an===q?0:an.startsWith(q)?1:an.includes(q)?2:3;
        const be=bn===q?0:bn.startsWith(q)?1:bn.includes(q)?2:3;
        return ae-be || an.localeCompare(bn,"it");
      }).slice(0,12);
  }
