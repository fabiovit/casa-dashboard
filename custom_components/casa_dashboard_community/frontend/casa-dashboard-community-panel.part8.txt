    if(name.includes("bedroom")) return "Camera";

    if(name.includes("balcony")) return "Balcone";

  _entityOptions() {
    if(!this._hass?.states) return [];
    return Object.entries(this._hass.states).map(([id,state])=>{
      const name=String(state?.attributes?.friendly_name || "").trim();
      return {id,name,label:name ? `${name} — ${id}` : id};
    }).sort((a,b)=>(a.name || a.id).localeCompare(b.name || b.id, "it", {sensitivity:"base"}));
  }
  _entityIdFromFriendlyName(value) {
    const raw=String(value || "").trim();
    if(!raw || raw.includes(".")) return raw;
    const normalized=raw.toLocaleLowerCase("it");
    const matches=this._entityOptions().filter(item=>item.name && item.name.toLocaleLowerCase("it")===normalized);
    return matches.length===1 ? matches[0].id : raw;
  }

  _renderConfigLayer() {
    let layer=this.shadowRoot.querySelector("#configLayer");
    if(!layer){ layer=document.createElement("div"); layer.id="configLayer"; this.shadowRoot.appendChild(layer); }
    layer.innerHTML=this._renderConfigDialog();
    const search=layer.querySelector("#configSearch");
    search?.addEventListener("input",ev=>{
      layer.querySelectorAll("[data-config-key]").forEach(input=>{ this._configDraft[input.dataset.configKey]=input.value; });
      this._configFilter=ev.target.value;
      this._renderConfigLayer();
      requestAnimationFrame(()=>{ const next=this.shadowRoot.querySelector("#configSearch"); next?.focus(); next?.setSelectionRange(next.value.length,next.value.length); });
    });
    layer.querySelectorAll("[data-config-key]").forEach(input=>{
      input.addEventListener("input",ev=>{ this._configDraft[ev.target.dataset.configKey]=ev.target.value; });
      input.addEventListener("change",ev=>{
        const resolved=this._entityIdFromFriendlyName(ev.target.value);
        if(resolved!==ev.target.value){ ev.target.value=resolved; this._configDraft[ev.target.dataset.configKey]=resolved; }
      });
    });

  async _saveConfigDialog() {
    const layer=this.shadowRoot.querySelector("#configLayer");
    layer?.querySelectorAll("[data-config-key]").forEach(input=>{ this._configDraft[input.dataset.configKey]=input.value; });
    const save=layer?.querySelector('[data-action="config-save"]');
    if(save){ save.disabled=true; save.innerHTML='<ha-icon icon="mdi:loading" class="spin"></ha-icon> Salvataggio…'; }
    try {
      const result=await this._hass.callWS({type:"casa_dashboard_community/config/save",entities:this._configDraft});
      this._entityMap=result?.entities || {...this._configDraft};
      this._entityConfigLoaded=true;
      this._configDialogOpen=false;
      this._renderConfigLayer();
      this._renderPage(false);
    } catch(err) {
      if(save){ save.disabled=false; save.innerHTML='<ha-icon icon="mdi:alert-circle-outline"></ha-icon> Riprova'; }
      alert(`Impossibile salvare la configurazione: ${err?.message || err}`);
    }
  }

  _state(id, fallback="—") {
    const pending=this._optimistic.get(id);
    if(pending && Date.now()-pending.at < 3500) return pending.state;
    const s=this._s(id)?.state;
    return (!s || ["unknown","unavailable","none",""].includes(s)) ? fallback : s;
  }

    const n=this._num(id,null); if(n===null) return "—";
