  _roomsForEntity(entity){
    return (this._effectiveRooms()||[]).filter(room =>
      Array.isArray(room?.entities) && room.entities.includes(entity)
    );
  }

  _overviewRoomTypesForEntity(entity){
    return this._roomsForEntity(entity).map(room=>String(room?.type||"").toLowerCase());
  }

  _overviewEntityIsIndoorRoom(entity){
    const rooms=this._roomsForEntity(entity);
    if(!rooms.length) return false;
    return rooms.some(room=>{
      const type=String(room?.type||"").toLowerCase();
      const name=String(room?.name||"").toLowerCase();
      return !["esterno","balcone"].includes(type)
        && !/esterno|outdoor|giardino|garden|terrazzo|balcone/.test(name);
    });
  }

  _overviewEntityIsOutdoorRoom(entity){
    const rooms=this._roomsForEntity(entity);
    if(!rooms.length) return false;
    return rooms.some(room=>{
      const type=String(room?.type||"").toLowerCase();
      const name=String(room?.name||"").toLowerCase();
      return ["esterno","balcone"].includes(type)
        || /esterno|outdoor|giardino|garden|terrazzo|balcone/.test(name);
    });
  }

  _overviewEntityLogicalKeys(entity){
    return this._logicalKeysForEntity(entity);
  }

  _overviewEntityHay(entity){
    const friendly=this._overviewLabelForEntity(entity);
    const keys=this._overviewEntityLogicalKeys(entity).join(" ");
    return `${entity} ${friendly} ${keys}`.toLowerCase();
  }

  _isOverviewTemperatureEntity(entity){
    const obj=this._hass?.states?.[entity];
    if(!obj) return false;
    const dc=String(obj.attributes?.device_class||"").toLowerCase();
    const unit=String(obj.attributes?.unit_of_measurement||"").toLowerCase();
    const direct=`${entity} ${this._overviewLabelForEntity(entity)}`.toLowerCase();

    if(dc==="humidity" || /umidit|humidity/.test(direct) || unit==="%") return false;
    if(/wallbox|evse|inverter|battery|batteria|board|scheda|case|pcb/.test(direct)) return false;
    return dc==="temperature" || /°c|°f/.test(unit) || /temperatura|temperature/.test(direct);
  }

  _isOverviewExternalEntity(entity){
    // Explicit room placement wins over the old 173-mapping compatibility layer.
    // An entity assigned to Cucina/Camera/etc. must never appear under Esterno
    // only because the same entity was once mapped to a weather logical key.
    if(this._overviewEntityIsIndoorRoom(entity)) return false;
    if(this._overviewEntityIsOutdoorRoom(entity)) return true;

    const hay=this._overviewEntityHay(entity);
    return /weather_|meteo|outdoor|esterno|vento|wind|pioggia|rain|uv|solar_lux|solar_radiation|radiazione/.test(hay)
      && !/wallbox|evse|vehicle|auto elettrica|ricarica/.test(hay);
  }

  _isOverviewSolarEntity(entity){
    const hay=this._overviewEntityHay(entity);
    const direct=`${entity} ${this._overviewLabelForEntity(entity)}`.toLowerCase();

    // Indoor room assignment excludes accidental matches coming only from a
    // legacy logical key. A real inverter/PV/battery entity is still allowed.
    if(this._overviewEntityIsIndoorRoom(entity)
      && !/solar|fotovolta|\bpv\b|battery|batteria|inverter|pcc/.test(direct)) return false;

    return /solar_|fotovolta|fotovoltaico|\bpv\b|pv[12]|battery|batteria|inverter|pcc|grid power|rete.*potenza|produzione.*fv/.test(hay)
      && !/wallbox|evse|vehicle|auto elettrica/.test(hay);
  }

  _isOverviewWallboxEntity(entity){
    const keys=this._overviewEntityLogicalKeys(entity);
    if(keys.some(k=>k.includes("vehicle_"))) return false;
    const hay=this._overviewEntityHay(entity);
    return /wallbox|evse|grid_current|ev_charging_at_home/.test(hay)
      || keys.some(k=>/wallbox|grid_current|ev_charging_at_home/.test(k));
  }

  _isOverviewMobilityEntity(entity){
    const hay=this._overviewEntityHay(entity);
    // Must match actual EV / Wallbox semantics. Generic room switches are excluded.
    const mobilityMatch=/wallbox|evse|\bev\b|electric vehicle|auto elettrica|vehicle|charging|ricarica|sessione.*energia|energia.*sessione|autonomia|range|odometer|contachilometri/.test(hay);
    if(!mobilityMatch) return false;

    // Explicitly reject room-appliance semantics even if friendly names are odd.
    if(/frigo|frigorif|lavastov|forno|microonde|cappa|airfryer|bimby|tv cucina|luce|lampada|led cucina|tenda cucina/.test(hay)) return false;

    const keys=this._overviewEntityLogicalKeys(entity);
    if(keys.some(k=>/wallbox|vehicle|ev_charging|charging_power|session_energy/.test(k))) return true;

    // Native domains commonly used by vehicle integrations may still be valid.
    return /wallbox|evse|\bev\b|vehicle|auto elettrica|charging|ricarica/.test(`${entity} ${this._overviewLabelForEntity(entity)}`.toLowerCase());
  }


  _roomClimateSummaryCards(){
    const rooms=(this._effectiveRooms()||[]).filter(room=>{
      const type=String(room?.type||"").toLowerCase();
      const name=String(room?.name||"").toLowerCase();
      return !["esterno","balcone"].includes(type)
        && !/esterno|outdoor|giardino|garden|terrazzo|balcone/.test(name);
    });

    const cards=[];
    for(const room of rooms){
      const ids=Array.isArray(room.entities)?room.entities:[];
      const tempCandidates=ids.filter(e=>{
        const obj=this._hass?.states?.[e]; if(!obj) return false;
        const dc=String(obj.attributes?.device_class||"").toLowerCase();
        const unit=String(obj.attributes?.unit_of_measurement||"").toLowerCase();
        const hay=`${e} ${this._roomEntityDisplay(e,room)}`.toLowerCase();
        return (dc==="temperature" || /°c|°f/.test(unit) || /temperatura|temperature/.test(hay))
          && !/clima|climatizz|setpoint|target|mandata|ritorno|return|wallbox|inverter|board|scheda|case/.test(hay);
      });
      const humCandidates=ids.filter(e=>{
        const obj=this._hass?.states?.[e]; if(!obj) return false;
        const dc=String(obj.attributes?.device_class||"").toLowerCase();
        const unit=String(obj.attributes?.unit_of_measurement||"").toLowerCase();
        const hay=`${e} ${this._roomEntityDisplay(e,room)}`.toLowerCase();
        return dc==="humidity" || unit==="%" || /umidit|humidity/.test(hay);
      });

      const temp=tempCandidates.find(e=>/ambiente|room|cucina|bagno|camera|garage|soggiorno|studio/.test(`${e} ${this._roomEntityDisplay(e,room)}`.toLowerCase()))
        || tempCandidates[0] || null;
      const hum=humCandidates[0]||null;
      if(!temp) continue;

      cards.push(`<div class="overview-room-climate-card">
        <span>${this._esc(room.name||"Stanza")}</span>
        <strong>${this._state(temp,"—")} ${this._attr(temp,"unit_of_measurement","")||""}</strong>
        <small>${hum?`${this._state(hum,"—")} ${this._attr(hum,"unit_of_measurement","")||""}`:""}</small>
      </div>`);
    }

    const outdoor=this._configuredEntitiesForOverview().filter(e=>this._isOverviewExternalEntity(e));
    const outTemp=outdoor.find(e=>this._isOverviewTemperatureEntity(e))||null;
    const outHum=outdoor.find(e=>{
      const dc=String(this._attr(e,"device_class","")).toLowerCase();
      const unit=String(this._attr(e,"unit_of_measurement","")).toLowerCase();
      const hay=`${e} ${this._overviewLabelForEntity(e)}`.toLowerCase();
      return dc==="humidity" || unit==="%" || /umidit|humidity/.test(hay);
    })||null;
    if(outTemp){
      cards.push(`<div class="overview-room-climate-card external">
        <span>Esterno</span>
        <strong>${this._state(outTemp,"—")} ${this._attr(outTemp,"unit_of_measurement","")||""}</strong>
        <small>${outHum?`${this._state(outHum,"—")} ${this._attr(outHum,"unit_of_measurement","")||""}`:""}</small>
      </div>`);
    }
    return cards.join("");
  }

  _roomCard(room) {
    if(!this._roomConfigured(room)) return "";
    const m=ROOM_META[room], s=this._roomStatus(room); let tags=[];
    if(room==="bagno") {
      if(this._configured("sensor.bathroom_temperature")) tags.push(`🌡️ ${this._fmt("sensor.bathroom_temperature",1,"°C")}`);
      if(this._configured("sensor.bathroom_humidity")) tags.push(`💧 ${this._fmt("sensor.bathroom_humidity",0,"%")}`);
    }
    if(room==="cucina") {
      if(this._configured("sensor.kitchen_temperature")) tags.push(`🌡️ ${this._fmt("sensor.kitchen_temperature",1,"°C")}`);
      if(this._configured("sensor.kitchen_humidity")) tags.push(`💧 ${this._fmt("sensor.kitchen_humidity",0,"%")}`);
    }
    if(room==="camera") {
      if(this._configured("sensor.bedroom_temperature")) tags.push(`🌡️ ${this._fmt("sensor.bedroom_temperature",1,"°C")}`);
      if(this._configured("sensor.bedroom_humidity")) tags.push(`💧 ${this._fmt("sensor.bedroom_humidity",0,"%")}`);
    }
    if(room==="balcone") {
      if(this._configured("sensor.weather_outdoor_temperature")) tags.push(`🌡️ ${this._fmt("sensor.weather_outdoor_temperature",1,"°C")}`);
      if(this._configured("sensor.weather_wind_gust")) tags.push(`💨 ${this._fmt("sensor.weather_wind_gust",0,"km/h")}`);
    }
    if(room==="garage") {
      if(this._configured("sensor.garage_temperature")) tags.push(`🌡️ ${this._fmt("sensor.garage_temperature",1,"°C")}`);
      if(this._configured("binary_sensor.ev_charging_at_home") || this._configured("sensor.wallbox_evse_status")) tags.push(this._on("binary_sensor.ev_charging_at_home")?"🔌 In carica":"🚗 Wallbox pronta");
    }
    if(room==="ingresso") {
      if(this._configured("alarm_control_panel.home_alarm")) tags.push(this._state("alarm_control_panel.home_alarm") === "disarmed"?"🔓 Allarme OFF":"🛡️ Allarme ON");
      if(this._configured("light.entry_main")) tags.push(this._on("light.entry_main")?"💡 Luce ON":"💡 Luce OFF");
    }
    return `<article class="card room-card premium-room clickable span-4" data-nav="${room}">
      <div class="room-heading">
        <div class="room-orb ${s.level}"><ha-icon icon="${m.icon}"></ha-icon></div>
        <div class="room-arrow"><ha-icon icon="mdi:arrow-top-right"></ha-icon></div>
      </div>
      <div class="room-copy">
        <div class="room-eyebrow">${m.subtitle}</div>
        <h3>${m.title}</h3>
        <div class="room-status-line ${s.level}">${this._esc(s.text)}</div>
      </div>
      <div class="room-pills">
        ${tags.map(t=>`<span class="room-pill">${t}</span>`).join("")}
      </div>
    </article>`;
  }
