          if(json?.overview && typeof json.overview==="object") overview=json.overview;

  _autoMapScore(candidate,key){
    const k=String(key||"").toLowerCase();
    const hay=candidate.hay;
    const direct=`${candidate.entity} ${candidate.name}`.toLowerCase();
    const domain=candidate.domain, dc=candidate.dc, unit=candidate.unit;
    let score=0;

    const add=(re,n)=>{ if(re.test(hay)) score+=n; };
    const words=k.replace(/^(sensor|binary_sensor|switch|input_boolean|alarm_control_panel)\./,"")
      .split(/[_\s]+/).filter(w=>w.length>2);
    for(const w of words) if(hay.includes(w)) score+=7;

    if(k.includes("temperature")){
      if(dc==="temperature") score+=55;
      if(/°c|°f/.test(unit)) score+=35;
      add(/temperatura|temperature/,25);
      if(dc==="humidity" || unit==="%") score-=100;
    }
    if(k.includes("humidity")){
      if(dc==="humidity") score+=55;
      if(unit==="%") score+=20;
      add(/umidit|humidity/,30);
      if(dc==="temperature") score-=25;
    }
    if(k.includes("illumin") || k.includes("lux")){
      if(dc==="illuminance") score+=55;
      if(unit==="lx") score+=40;
      add(/lux|luminos|illumin/,28);
    }
    if(k.includes("motion") || k.includes("occupancy") || k.includes("presence")){
      if(["motion","occupancy","presence"].includes(dc)) score+=60;
      add(/movimento|motion|occupaz|presenza|presence/,30);
    }
    if(k.includes("door") || k.includes("window") || k.includes("opening")){
      if(["door","window","opening","garage_door"].includes(dc)) score+=60;
      add(/porta|door|finestra|window|apertura/,28);
    }
    if(k.includes("climate")){
      if(domain==="climate") score+=75;
      add(/clima|climatizz|condizion|air.?condition/,30);
    }
    if(k.includes("light")){
      if(domain==="light") score+=70;
      add(/luce|light|lamp|led|plafon/,24);
    }
    if(k.includes("alarm")){
      if(domain==="alarm_control_panel") score+=90;
      add(/allarme|alarm/,35);
    }

    if(/wallbox|ev_|vehicle/.test(k)){
      if(/wallbox|evse|auto elettrica|vehicle|ricarica|charging/.test(direct)) score+=70;
      if(/frigo|forno|lavastov|microonde|cappa|luce|cucina/.test(direct)) score-=120;
    }
    if(/solar|pv|inverter|battery|grid/.test(k)){
      if(/solar|fotovolta|\bpv\b|inverter|battery|batteria|pcc|grid|rete/.test(direct)) score+=65;
      if(/wallbox|evse/.test(direct) && !/grid/.test(k)) score-=50;
    }
    if(k.includes("weather")){
      if(/meteo|weather|outdoor|esterno|vento|wind|pioggia|rain|radiazione|\buv\b/.test(direct)) score+=60;
    }

    if(candidate.room){
      const roomText=`${candidate.room.name||""} ${candidate.room.type||""}`.toLowerCase();
      const roomRules=[
        [/kitchen|cucina/,/cucina|kitchen/],
        [/bathroom|bagno/,/bagno|bathroom/],
        [/bedroom|camera/,/camera|bedroom/],
        [/garage/,/garage/],
        [/entry|entrance|ingresso/,/ingresso|entry|entrance/],
        [/hallway|disimpegno|corridoio/,/disimpegno|corridoio|hallway|corridor/],
        [/balcony|balcone/,/balcone|terrazzo|balcony/],
      ];
      for(const [keyRe,roomRe] of roomRules) if(keyRe.test(k) && roomRe.test(roomText)) score+=50;
    }

    return score;
  }

  _autoMapBestCandidate(key,candidates,used){
    let best=null,bestScore=0;
    for(const c of candidates){
      if(used.has(c.entity)) continue;
      const score=this._autoMapScore(c,key);
      if(score>bestScore){ best=c; bestScore=score; }
    }
    return bestScore>=45 ? {...best,_score:bestScore} : null;
  }

  _autoMapEmptyMappings(){
    const roomCandidates=this._autoMapRoomCandidates();
    const globalCandidates=this._autoMapGlobalCandidates();
    const used=new Set(Object.values(this._configDraft||{}).map(v=>String(v||"").trim()).filter(Boolean));
    const changes=[];

    for(const key of ENTITY_KEYS){
      if(String(this._configDraft?.[key]||"").trim()) continue;

      let candidate=this._autoMapBestCandidate(key,roomCandidates,used);
      if(!candidate) candidate=this._autoMapBestCandidate(key,globalCandidates,used);
      if(!candidate) continue;

      this._configDraft[key]=candidate.entity;
      used.add(candidate.entity);
      changes.push({key,entity:candidate.entity,score:candidate._score});
    }

    this._autoMapChanges=changes;
    this._autoMapNotice=changes.length
      ? `Auto-abbinamento: ${changes.length} campi vuoti compilati. Controlla i risultati e poi premi Salva configurazione.`
      : "Auto-abbinamento: nessuna nuova associazione sufficientemente sicura trovata.";
    this._renderConfigLayer();
  }

  _resolveRoomEntityToken(token){
    const raw=String(token||"").trim();
    if(!raw) return "";
    if(this._hass?.states?.[raw]) return raw;
    const needle=raw.toLocaleLowerCase("it");
    const states=Object.entries(this._hass?.states||{});

    const exactFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it")===needle
    );
    if(exactFriendly.length===1) return exactFriendly[0][0];

    const exactTail=states.filter(([id]) =>
      id.split(".").slice(1).join(".").replaceAll("_"," ").toLocaleLowerCase("it")===needle
    );
    if(exactTail.length===1) return exactTail[0][0];

    const partialFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it").includes(needle)
    );
    if(partialFriendly.length===1) return partialFriendly[0][0];

    return raw;
  }

  async _readImageFile(file){
    if(!file) throw new Error("Seleziona un file immagine.");
    const type=String(file.type||"").toLowerCase();
    const name=String(file.name||"").toLowerCase();
    const imageLike=type.startsWith("image/") || /\.(jpe?g|png|webp|heic|heif)$/i.test(name);
    if(!imageLike) throw new Error("Formato non supportato. Usa JPG, PNG, WebP, HEIC o HEIF.");
    if(file.size>25*1024*1024) throw new Error("La foto originale è troppo grande (max 25 MB).");

    if("createImageBitmap" in window){
      try{
        const bitmap=await createImageBitmap(file,{imageOrientation:"from-image"});
        return {
          source:bitmap,
          naturalWidth:bitmap.width,
          naturalHeight:bitmap.height,
          width:bitmap.width,
          height:bitmap.height,
          close:()=>bitmap.close?.()
        };
      }catch(_err){}
    }

    const source=await new Promise((resolve,reject)=>{
      const reader=new FileReader();
      reader.onload=()=>resolve(reader.result);
      reader.onerror=()=>reject(new Error("Impossibile leggere la foto."));
      reader.readAsDataURL(file);
    });
    try{
      const img=await new Promise((resolve,reject)=>{
        const el=new Image();
        el.onload=()=>resolve(el);
        el.onerror=()=>reject(new Error("decode"));
        el.src=source;
      });
      img.source=img;
      img.close=()=>{};
      return img;
    }catch(_err){
      if(type.includes("heic") || type.includes("heif") || /\.(heic|heif)$/i.test(name)){
        throw new Error("Questa foto HEIC/HEIF non è supportata dal browser in uso. Salvala come JPG/PNG e riprova.");
      }
      throw new Error("Il browser non riesce a decodificare questa immagine. Usa JPG, PNG o WebP.");
    }
  }

    const img=await this._readImageFile(file);

    const drawSource=img.source||img;
    if(!naturalW || !naturalH) throw new Error("Dimensioni immagine non valide.");
    let side=Math.min(maxSide,Math.max(naturalW,naturalH));
    let quality=.82;
    let best="";
    for(le
