t:"Raffiche forti"};

  _renderNav(force=false) {
    const nav=this.shadowRoot.querySelector("#nav"); if(!nav) return;
    const visibleNav=NAV.filter(([id])=>id==="panoramica" || this._roomConfigured(id));
    if(this._active!=="panoramica" && !this._roomConfigured(this._active)) this._active="panoramica";
    if(force || !nav.children.length || nav.children.length!==visibleNav.length){
      nav.innerHTML=visibleNav.map(([id,icon,label])=>`<button class="nav-btn ${id===this._active?"active":""}" data-nav="${id}" type="button"><ha-icon icon="${icon}"></ha-icon>${label}</button>`).join("");
    } else {
      nav.querySelectorAll(".nav-btn").forEach(btn=>btn.classList.toggle("active",btn.dataset.nav===this._active));
    }
  }

    if(!this._hass || !this.shadowRoot) return;
    this._renderNav(forceNav); this._renderHeader(); this._applyThemeMode();
    const host=this.shadowRoot.querySelector("#content"); if(!host) return;

    host.innerHTML=renderer ? renderer() : `<div class="empty">Pagina non disponibile</div>`;
    this._cleanEmptySections(host);
  }

  _cleanEmptySections(host){
    host.querySelectorAll(".grid").forEach(grid=>{
      if(!grid.children.length){
        const prev=grid.previousElementSibling;
        if(prev?.classList?.contains("chapter")) prev.remove();
        grid.remove();
      }
    });
    const mosaic=host.querySelector(".room-mosaic");
    if(mosaic && !mosaic.children.length){
      const prev=mosaic.previousElementSibling;
      if(prev?.classList?.contains("chapter")) prev.remove();
      mosaic.remove();
    }
  }
