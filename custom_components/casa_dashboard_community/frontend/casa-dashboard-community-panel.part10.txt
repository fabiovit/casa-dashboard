  async _loadEntityConfig(force=false) {
    if((this._entityConfigLoaded && !force) || this._entityConfigLoading) return;
    this._entityConfigLoading=true;
    try {
      let entities=null;
      let labels=null;
      let overview=null;
      if(this._hass?.callWS){
        try {
          const result=await this._hass.callWS({type:"casa_dashboard_community/config/get"});
          if(result?.entities && typeof result.entities==="object") entities=result.entities;
          if(result?.labels && typeof result.labels==="object") labels=result.labels;
          if(result?.overview && typeof result.overview==="object") overview=result.overview;
          if(Array.isArray(result?.rooms)) this._rooms=result.rooms;
        } catch(_wsErr) { /* fallback below */ }
      }
      if(!entities){
        const res=await fetch(`${ENTITY_CONFIG_URL}?v=${Date.now()}`, {cache:"no-store"});
        if(res.ok){
          const json=await res.json();
          if(json?.entities && typeof json.entities==="object") entities=json.entities;
          if(json?.labels && typeof json.labels==="object") labels=json.labels;
          if(json?.overview && typeof json.overview==="object") overview=json.overview;
        }
      }
      this._entityMap=entities || {};
      this._entityLabels=labels || {};
      this._overviewConfig=overview || {};
    } catch(_err) {
      this._entityMap={};
      this._entityLabels={};
      this._overviewConfig={};
    } finally {
      this._entityConfigLoaded=true;
      this._entityConfigLoading=false;
      this._renderPage(false);
    }
  }

  _resolveRoomEntityToken(token){
    const raw=String(token||"").trim();
    if(!raw) return "";
    if(this._hass?.states?.[raw]) return raw;
    const needle=raw.toLocaleLowerCase("it");
    const states=Object.entries(this._hass?.states||{});

    const exactFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it")===needle
    );
    if(exactFriendly.length===1) return exactFriendly[0][0];

    const exactTail=states.filter(([id]) =>
      id.split(".").slice(1).join(".").replaceAll("_"," ").toLocaleLowerCase("it")===needle
    );
    if(exactTail.length===1) return exactTail[0][0];

    const partialFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it").includes(needle)
    );
    if(partialFriendly.length===1) return partialFriendly[0][0];

    return raw;
  }

  async _resizeImageForUpload(file,maxSide=1400,quality=.84){
    if(!file || !String(file.type||"").startsWith("image/")) throw new Error("Seleziona un file immagine.");
    if(file.size>12*1024*1024) throw new Error("La foto originale è troppo grande (max 12 MB).");
    const source=await new Promise((resolve,reject)=>{
      const reader=new FileReader();
      reader.onload=()=>resolve(reader.result);
      reader.onerror=()=>reject(new Error("Impossibile leggere la foto."));
      reader.readAsDataURL(file);
    });
    const img=await new Promise((resolve,reject)=>{
      const el=new Image();
      el.onload=()=>resolve(el);
      el.onerror=()=>reject(new Error("Formato immagine non valido."));
      el.src=source;
    });
    const longest=Math.max(img.naturalWidth||img.width,img.naturalHeight||img.height);
    const scale=Math.min(1,maxSide/Math.max(1,longest));
    const width=Math.max(1,Math.round((img.naturalWidth||img.width)*scale));
    const height=Math.max(1,Math.round((img.naturalHeight||img.height)*scale));
    const canvas=document.createElement("canvas");
    canvas.width=width; canvas.height=height;
    const ctx=canvas.getContext("2d");
    if(!ctx) throw new Error("Il browser non può elaborare la foto.");
    ctx.drawImage(img,0,0,width,height);
    return canvas.toDataURL("image/jpeg",quality);
  }

  _syncConfiguredKeyToRoom(key,oldEntity,newEntity){
    // Legacy 173 mappings are used only to bootstrap the room list once.
    // After rooms have been saved, room membership is controlled exclusively
    // from the Rooms manager and must never be recreated automatically.
    if(Array.isArray(this._rooms) && this._rooms.length) return;
    const roomId=this._legacyRoomForConfigKey(key);
    if(!roomId) return;
    const defaults=new Map(this._defaultRooms().map(r=>[r.id,r]));
    let room=this._roomsDraft.find(r=>r.id===roomId) || this._roomsDraft.find(r=>r.type===roomId);
    if(!room && newEntity){
      const meta=defaults.get(roomId);
      if(meta){ room={...meta,entities:[],entity_labels:{},entity_images:{}}; this._roomsDraft.push(room); }
    }
    if(!room) return;
    room.entities=Array.isArray(room.entities)?room.entities:[];
    if(oldEntity && oldEntity!==newEntity){
      const stillUsed=[...ENTITY_KEYS].some(k=>
        k!==key &&
        this._legacyRoomForConfigKey(k)===roomId &&
        String(this._configDraft[k]||"").trim()===oldEntity
      );
      if(!stillUsed) room.entities=room.entities.filter(e=>e!==oldEntity);
    }
    if(newEntity && !room.entities.includes(newEntity)) room.entities.push(newEntity);
  }

  _slugRoom(name){
    const base=String(name||"stanza").normalize("NFD").replace(/[\u0300-\u036f]/g,"").toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-|-$/g,"")||"stanza";
    let id=`room-${base}`, n=2; const used=new Set(this._roomsDraft.map(r=>r.id));
    while(used.has(id)) id=`room-${base}-${n++}`; return id;
  }

      input?.focus(); input?.select();
    });
  }
