  async _readImageFile(file){
    if(!file) throw new Error("Seleziona un file immagine.");
    const type=String(file.type||"").toLowerCase();
    const name=String(file.name||"").toLowerCase();
    const imageLike=type.startsWith("image/") || /\.(jpe?g|png|webp|heic|heif)$/i.test(name);
    if(!imageLike) throw new Error("Formato non supportato. Usa JPG, PNG o WebP.");
    if(file.size>20*1024*1024) throw new Error("La foto originale è troppo grande (max 20 MB).");
    const source=await new Promise((resolve,reject)=>{
      const reader=new FileReader();
      reader.onload=()=>resolve(reader.result);
      reader.onerror=()=>reject(new Error("Impossibile leggere la foto."));
      reader.readAsDataURL(file);
    });
    try{
      return await new Promise((resolve,reject)=>{
        const el=new Image();
        el.onload=()=>resolve(el);
        el.onerror=()=>reject(new Error("decode"));
        el.src=source;
      });
    }catch(_err){
      if(type.includes("heic") || type.includes("heif") || /\.(heic|heif)$/i.test(name)){
        throw new Error("Questa foto è HEIC/HEIF e il browser non riesce a convertirla. Esportala come JPG e riprova.");
      }
      throw new Error("Il browser non riesce a decodificare questa immagine. Usa JPG, PNG o WebP.");
    }
  }
  async _compressImageForUpload(file,{maxSide=1200,targetBytes=700000,minSide=640}={}){
    const img=await this._readImageFile(file);
    const naturalW=img.naturalWidth||img.width;
    const naturalH=img.naturalHeight||img.height;
    if(!naturalW || !naturalH) throw new Error("Dimensioni immagine non valide.");
    let side=Math.min(maxSide,Math.max(naturalW,naturalH));
    let quality=.82;
    let best="";
    for(let attempt=0;attempt<9;attempt++){
      const scale=Math.min(1,side/Math.max(naturalW,naturalH));
      const width=Math.max(1,Math.round(naturalW*scale));
      const height=Math.max(1,Math.round(naturalH*scale));
      const canvas=document.createElement("canvas");
      canvas.width=width; canvas.height=height;
      const ctx=canvas.getContext("2d");
      if(!ctx) throw new Error("Il browser non può elaborare la foto.");
      ctx.fillStyle="#ffffff";
      ctx.fillRect(0,0,width,height);
      ctx.drawImage(img,0,0,width,height);
      const encoded=canvas.toDataURL("image/jpeg",quality);
      best=encoded;
      if(this._dataUrlBytes(encoded)<=targetBytes) return encoded;
      if(quality>.58) quality-=.08;
      else side=Math.max(minSide,Math.round(side*.84));
    }
    if(best && this._dataUrlBytes(best)<=targetBytes*1.15) return best;
    throw new Error(`Non riesco a comprimere la foto sotto ${Math.round(targetBytes/1000)} KB. Prova una foto più piccola.`);
  }

  async _uploadDashboardImage(file,onInlineReady=null){
    // First create a small inline image and expose it immediately to the UI.
    // This guarantees an instant preview even if backend file upload is slow/unavailable.
    const inline=await this._compressImageForUpload(file,{maxSide:820,targetBytes:260000,minSide:480});
    if(!inline || this._dataUrlBytes(inline)>300000){
      throw new Error("La foto compressa è ancora troppo grande. Prova una foto più piccola.");
    }
    if(typeof onInlineReady==="function") onInlineReady(inline);

    // Then try to store a slightly larger copy as a normal /local file.
    // If anything goes wrong or takes too long, keep the already-working inline image.
    try{
      const image=await this._compressImageForUpload(file,{maxSide:1200,targetBytes:650000,minSide:640});
      const uploadPromise=this._hass.callWS({type:"casa_dashboard_community/config/upload_image",image});
      const timeoutPromise=new Promise((_,reject)=>setTimeout(()=>reject(new Error("Timeout upload foto")),7000));
      const result=await Promise.race([uploadPromise,timeoutPromise]);
      if(result?.url){
        const verified=await Promise.race([
          this._verifyDashboardImageUrl(result.url),
          new Promise(resolve=>setTimeout(()=>resolve(false),3500))
        ]);
        if(verified) return {url:result.url,mode:"file",inline};
      }
    }catch(err){
      console.warn("[Casa Dashboard Community] backend image upload unavailable; inline image kept",err);
    }
    return {url:inline,mode:"inline",inline};
  }

  _syncConfiguredKeyToRoom(key,oldEntity,newEntity){
    // Legacy 173 mappings are used only to bootstrap the room list once.
    // After rooms have been saved, room membership is controlled exclusively
    // from the Rooms manager and must never be recreated automatically.
    if(Array.isArray(this._rooms) && this._rooms.length) return;
    const roomId=this._legacyRoomForConfigKey(key);

    const defaults=new Map(this._defaultRooms().map(r=>[r.id,r]));
    let room=this._roomsDraft.find(r=>r.id===roomId) || this._roomsDraft.find(r=>r.type===roomId);
    if(!room && newEntity){
      const meta=defaults.get(roomId);
      if(meta){ room={...meta,entities:[],entity_labels:{},entity_images:{}}; this._roomsDraft.push(room); }
    }
    if(!room) return;
    room.entities=Array.isArray(room.entities)?room.entities:[];
    if(oldEntity && oldEntity!==newEntity){
      const stillUsed=[...ENTITY_KEYS].some(k=>
        k!==key &&
        this._legacyRoomForConfigKey(k)===roomId &&
        String(this._configDraft[k]||"").trim()===oldEntity
      );
      if(!stillUsed) room.entities=room.entities.filter(e=>e!==oldEntity);
    }

  _slugRoom(name){
    const base=String(name||"stanza").normalize("NFD").replace(/[\u0300-\u036f]/g,"").toLowerCase().replace(/[^a-z0-9]+/g,"-").replace(/^-|-$/g,"")||"stanza";
    let id=`room-${base}`, n=2; const used=new Set(this._roomsDraft.map(r=>r.id));
    while(used.has(id)) id=`room-${base}-${n++}`; return id;
  }
