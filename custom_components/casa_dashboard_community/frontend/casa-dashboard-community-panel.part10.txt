  _autoMapRoomCandidates(){
    const out=[];
    for(const room of (this._effectiveRooms()||[])){
      for(const entity of (room.entities||[])){
        const obj=this._hass?.states?.[entity];
        if(!obj) continue;
        const name=this._roomEntityDisplay(entity,room);
        out.push({
          entity,room,name,
          dc:String(obj.attributes?.device_class||"").toLowerCase(),
          unit:String(obj.attributes?.unit_of_measurement||"").toLowerCase(),
          domain:String(entity).split(".")[0],
          hay:`${entity} ${name} ${room.name||""} ${room.type||""}`.toLowerCase()
        });
      }
    }
    return out;
  }

  _autoMapGlobalCandidates(){
    return Object.entries(this._hass?.states||{}).map(([entity,obj])=>({
      entity,
      room:null,
      name:String(obj?.attributes?.friendly_name||entity),
      dc:String(obj?.attributes?.device_class||"").toLowerCase(),
      unit:String(obj?.attributes?.unit_of_measurement||"").toLowerCase(),
      domain:String(entity).split(".")[0],
      hay:`${entity} ${obj?.attributes?.friendly_name||""}`.toLowerCase()
    }));
  }

  _autoMapScore(candidate,key){
    const k=String(key||"").toLowerCase();
    const hay=candidate.hay;
    const direct=`${candidate.entity} ${candidate.name}`.toLowerCase();
    const domain=candidate.domain, dc=candidate.dc, unit=candidate.unit;
    let score=0;

    const add=(re,n)=>{ if(re.test(hay)) score+=n; };
    const words=k.replace(/^(sensor|binary_sensor|switch|input_boolean|alarm_control_panel)\./,"")
      .split(/[_\s]+/).filter(w=>w.length>2);
    for(const w of words) if(hay.includes(w)) score+=7;

    if(k.includes("temperature")){
      if(dc==="temperature") score+=55;
      if(/°c|°f/.test(unit)) score+=35;
      add(/temperatura|temperature/,25);
      if(dc==="humidity" || unit==="%") score-=100;
    }
    if(k.includes("humidity")){
      if(dc==="humidity") score+=55;
      if(unit==="%") score+=20;
      add(/umidit|humidity/,30);
      if(dc==="temperature") score-=25;
    }
    if(k.includes("illumin") || k.includes("lux")){
      if(dc==="illuminance") score+=55;
      if(unit==="lx") score+=40;
      add(/lux|luminos|illumin/,28);
    }
    if(k.includes("motion") || k.includes("occupancy") || k.includes("presence")){
      if(["motion","occupancy","presence"].includes(dc)) score+=60;
      add(/movimento|motion|occupaz|presenza|presence/,30);
    }
    if(k.includes("door") || k.includes("window") || k.includes("opening")){
      if(["door","window","opening","garage_door"].includes(dc)) score+=60;
      add(/porta|door|finestra|window|apertura/,28);
    }
    if(k.includes("climate")){
      if(domain==="climate") score+=75;
      add(/clima|climatizz|condizion|air.?condition/,30);
    }
    if(k.includes("light")){
      if(domain==="light") score+=70;
      add(/luce|light|lamp|led|plafon/,24);
    }
    if(k.includes("alarm")){
      if(domain==="alarm_control_panel") score+=90;
      add(/allarme|alarm/,35);
    }

    if(/wallbox|ev_|vehicle/.test(k)){
      if(/wallbox|evse|auto elettrica|vehicle|ricarica|charging/.test(direct)) score+=70;
      if(/frigo|forno|lavastov|microonde|cappa|luce|cucina/.test(direct)) score-=120;
    }
    if(/solar|pv|inverter|battery|grid/.test(k)){
      if(/solar|fotovolta|\bpv\b|inverter|battery|batteria|pcc|grid|rete/.test(direct)) score+=65;
      if(/wallbox|evse/.test(direct) && !/grid/.test(k)) score-=50;
    }
    if(k.includes("weather")){
      if(/meteo|weather|outdoor|esterno|vento|wind|pioggia|rain|radiazione|\buv\b/.test(direct)) score+=60;
    }

    if(candidate.room){
      const roomText=`${candidate.room.name||""} ${candidate.room.type||""}`.toLowerCase();
      const roomRules=[
        [/kitchen|cucina/,/cucina|kitchen/],
        [/bathroom|bagno/,/bagno|bathroom/],
        [/bedroom|camera/,/camera|bedroom/],
        [/garage/,/garage/],
        [/entry|entrance|ingresso/,/ingresso|entry|entrance/],
        [/hallway|disimpegno|corridoio/,/disimpegno|corridoio|hallway|corridor/],
        [/balcony|balcone/,/balcone|terrazzo|balcony/],
      ];
      for(const [keyRe,roomRe] of roomRules) if(keyRe.test(k) && roomRe.test(roomText)) score+=50;
    }

    return score;
  }

  _autoMapBestCandidate(key,candidates,used){
    let best=null,bestScore=0;
    for(const c of candidates){
      if(used.has(c.entity)) continue;
      const score=this._autoMapScore(c,key);
      if(score>bestScore){ best=c; bestScore=score; }
    }
    return bestScore>=45 ? {...best,_score:bestScore} : null;
  }

  _autoMapEmptyMappings(){
    const roomCandidates=this._autoMapRoomCandidates();
    const globalCandidates=this._autoMapGlobalCandidates();
    const used=new Set(Object.values(this._configDraft||{}).map(v=>String(v||"").trim()).filter(Boolean));
    const changes=[];

    for(const key of ENTITY_KEYS){
      if(String(this._configDraft?.[key]||"").trim()) continue;

      let candidate=this._autoMapBestCandidate(key,roomCandidates,used);
      if(!candidate) candidate=this._autoMapBestCandidate(key,globalCandidates,used);
      if(!candidate) continue;

      this._configDraft[key]=candidate.entity;
      used.add(candidate.entity);
      changes.push({key,entity:candidate.entity,score:candidate._score});
    }

    this._autoMapChanges=changes;
    this._autoMapNotice=changes.length
      ? `Auto-abbinamento: ${changes.length} campi vuoti compilati. Controlla i risultati e poi premi Salva configurazione.`
      : "Auto-abbinamento: nessuna nuova associazione sufficientemente sicura trovata.";
    this._renderConfigLayer();
  }

  _resolveRoomEntityToken(token){
    const raw=String(token||"").trim();
    if(!raw) return "";
    if(this._hass?.states?.[raw]) return raw;
    const needle=raw.toLocaleLowerCase("it");
    const states=Object.entries(this._hass?.states||{});

    const exactFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it")===needle
    );
    if(exactFriendly.length===1) return exactFriendly[0][0];

    const exactTail=states.filter(([id]) =>
      id.split(".").slice(1).join(".").replaceAll("_"," ").toLocaleLowerCase("it")===needle
    );
    if(exactTail.length===1) return exactTail[0][0];

    const partialFriendly=states.filter(([id,obj]) =>
      String(obj?.attributes?.friendly_name||"").trim().toLocaleLowerCase("it").includes(needle)
    );
    if(partialFriendly.length===1) return partialFriendly[0][0];

    return raw;
  }

  _dataUrlBytes(dataUrl){
    const encoded=String(dataUrl||"").split(",")[1]||"";

  async _readImageFile(file){
    if(!file) throw new Error("Seleziona un file immagine.");
    const type=String(file.type||"").toLowerCase();
    const name=String(file.name||"").toLowerCase();
    const imageLike=type.startsWith("image/") || /\.(jpe?g|png|webp|heic|heif)$/i.test(name);
    if(!imageLike) throw new Error("Formato non supportato. Usa JPG, PNG, WebP, HEIC o HEIF.");
    if(file.size>25*1024*1024) throw new Error("La foto originale è troppo grande (max 25 MB).");

    if("createImageBitmap" in window){
      try{
        const bitmap=await createImageBitmap(file,{imageOrientation:"from-image"});
        return {
          source:bitmap,
          naturalWidth:bitmap.width,
          naturalHeight:bitmap.height,
          width:bitmap.width,
          height:bitmap.height,
          close:()=>bitmap.close?.()
        };
      }catch(_err){}
    }

    const source=await new Promise((resolve,reject)=>{
      const reader=new FileReader();
      reader.onload=()=>resolve(reader.result);
      reader.onerror=()=>reject(new Error("Impossibile leggere la foto."));
      reader.readAsDataURL(file);
    });
    try{
      const img=await new Promise((resolve,reject)=>{
        const el=new Image();
        el.onload=()=>resolve(el);
        el.onerror=()=>reject(new Error("decode"));
        el.src=source;
      });
      img.source=img;
      img.close=()=>{};
      return img;
    }catch(_err){
      if(type.includes("heic") || type.includes("heif") || /\.(heic|heif)$/i.test(name)){
        throw new Error("Questa foto HEIC/HEIF non è supportata dal browser in uso. Salvala come JPG/PNG e riprova.");
      }
      throw new Error("Il browser non riesce a decodificare questa immagine. Usa JPG, PNG o WebP.");
    }
  }
  async _compressImageForUpload(file,{maxSide=1200,targetBytes=700000,minSide=640}={}){
    const img=await this._readImageFile(file);
    const naturalW=img.naturalWidth||img.width;
    const naturalH=img.naturalHeight||img.height;
    const drawSource=img.source||img;
    if(!naturalW || !naturalH) throw new Error("Dimensioni immagine non valide.");
    let side=Math.min(maxSide,Math.max(naturalW,naturalH));
    let quality=.82;
    let best="";
    for(let attempt=0;attempt<9;attempt++){
      const scale=Math.min(1,side/Math.max(naturalW,naturalH));
      const width=Math.max(1,Math.round(naturalW*scale));
      const height=Math.max(1,Math.round(naturalH*scale));
      const canvas=document.createElement("canvas");
      canvas.width=width; canvas.height=height;
      const ctx=canvas.getContext("2d");
      if(!ctx) throw new Error("Il browser non può elaborare la foto.");
      ctx.fillStyle="#ffffff";
      ctx.fillRect(0,0,width,height);
      ctx.drawImage(drawSource,0,0,width,height);
      const encoded=canvas.toDataURL("image/jpeg",quality);
      best=encoded;
      if(this._dataUrlBytes(encoded)<=targetBytes){ img.close?.(); return encoded; }
      if(quality>.58) quality-=.08;
      else side=Math.max(minSide,Math.round(side*.84));
    }
    if(best && this._dataUrlBytes(best)<=targetBytes*1.15) return best;
    throw new Error(`Non riesco a comprimere la foto sotto ${Math.round(targetBytes/1000)} KB. Prova una foto più piccola.`);
  }
  async _verifyDashboardImageUrl(url){
    if(!url || !String(url).startsWith("/local/")) return false;
    try{
      const joiner=url.includes("?")?"&":"?";
      const response=await fetch(`${url}${joiner}cdc=${Date.now()}`,{method:"GET",cache:"no-store"});
      return !!response?.ok;
    }catch(_err){
      return false;
    }
  }
  async _uploadDashboardImage(file,onInlineReady=null){
    // First create a small inline image and expose it immediately to the UI.
    // This guarantees an instant preview even if backend file upload is slow/unavailable.
    const inline=await this._compressImageForUpload(file,{maxSide:820,targetBytes:260000,minSide:480});
    if(!inline || this._dataUrlBytes(inline)>300000){
      throw new Error("La foto compressa è ancora troppo grande. Prova una foto più piccola.");
    }
    if(typeof onInlineReady==="function") onInlineReady(inline);

    // Then try to store a slightly larger copy as a normal /local file.
    // If anything goes wrong or takes too long, keep the already-working inline image.
    try{
      const image=await this._compressImageForUpload(file,{maxSide:1200,targetBytes:650000,minSide:640});
      const uploadPromise=this._hass.callWS({type:"casa_dashboard_community/config/upload_image",image});
      const timeoutPromise=new Promise((_,reject)=>setTimeout(()=>reject(new Error("Timeout upload foto")),7000));
      const result=await Promise.race([uploadPromise,timeoutPromise]);
      if(result?.url){
        const verified=await Promise.race([
          this._verifyDashboardImageUrl(result.url),
          new Promise(resolve=>setTimeout(()=>resolve(false),3500))
        ]);
        if(verified) return {url:result.url,mode:"file",inline};
      }
    }catch(err){
      console.warn("[Casa Dashboard Community] backend image upload unavailable; inline image kept",err);
    }
    return {url:inline,mode:"inline",inline};
  }

  _mergeConfiguredEntitiesIntoRooms(rooms){
    const out=(Array.isArray(rooms)?rooms:[]).map(r=>({...r,entities:[...(r.entities||[])],entity_labels:{...(r.entity_labels||{})},entity_images:{...(r.entity_images||{})},entity_icons:{...(r.entity_icons||{})},entity_sizes:{...(r.entity_sizes||{})}}));
    const defaults=new Map(this._defaultRooms().map(r=>[r.id,r]));
    const configured=[...ENTITY_KEYS].map(key=>{
      const entity=String(this._configDraft?.[key] ?? this._entityMap?.[key] ?? "").trim();
      return {key,entity,room:this._legacyRoomForConfigKey(key)};
    }).filter(x=>x.entity && x.room);

    for(const item of configured){
      const meta=defaults.get(item.room);
      let room=out.find(r=>r.id===item.room)
        || out.find(r=>r.type===item.room)
        || out.find(r=>String(r.name||"").toLocaleLowerCase("it")===String(meta?.name||"").toLocaleLowerCase("it"));
      if(!room && meta){ room={...meta,entities:[],entity_labels:{},entity_images:{}}; out.push(room); }
      if(room){
        room.entities=Array.isArray(room.entities)?room.entities:[];
        if(!room.entities.includes(item.entity)) room.entities.push(item.entity);
      }
    }
    return out;
  }
  _syncConfiguredKeyToRoom(key,oldEntity,newEntity){
    // Legacy 173 mappings are used only to bootstrap the room list once.
    // After rooms have been saved, room membership is controlled exclusively
    // from the Rooms manager and must never be recreated automatically.
    if(Array.isArray(this._rooms) && this._rooms.length) return;
    const roomId=this._legacyRoomForConfigKey(key);
    if(!roomId) return;
    const defaults=new Map(this._defaultRooms().map(r=>[r.id,r]));
    let room=this._roomsDraft.find(r=>r.id===roomId) || this._roomsDraft.find(r=>r.type===roomId);
    if(!room && newEntity){
      const meta=defaults.get(roomId);
      if(meta){ room={...meta,entities:[],entity_labels:{},entity_images:{}}; this._roomsDraft.push(room); }
    }
    if(!room) return;
    room.entities=Array.isArray(room.entities)?room.entities:[];
    if(oldEntity && oldEntity!==newEntity){
      const stillUsed=[...ENTITY_KEYS].some(k=>
        k!==key &&
        this._legacyRoomForConfigKey(k)===roomId &&
        String(this._configDraft[k]||"").trim()===oldEntity
      );
      if(!stillUsed) room.entities=room.entities.filter(e=>e!==oldEntity);
    }
    if(newEntity && !room.entities.includes(newEntity)) room.entities.push(newEntity);
  }
