  _fmtPowerAuto(idOrIds, digits=2){
    const ids = Array.isArray(idOrIds) ? idOrIds : [idOrIds];
    const id = this._firstId(ids);
    if(!id) return "—";
    const n = this._num(id, null);
    if(n===null) return "—";
    const rawUnit = String(this._attr(id, "unit_of_measurement", "") || "").toLowerCase();
    if(rawUnit === 'w' || (!rawUnit && Math.abs(n) >= 100)) return `${(n/1000).toFixed(digits)} kW`;
    if(rawUnit === 'kw') return `${n.toFixed(digits)} kW`;
    return `${n.toFixed(digits)} ${rawUnit || 'kW'}`;
  }
  _powerValueAuto(idOrIds){
    const ids = Array.isArray(idOrIds) ? idOrIds : [idOrIds];
    const id = this._firstId(ids);
    if(!id) return null;
    let n = this._num(id, null);
    if(n===null) return null;
    const rawUnit = String(this._attr(id, "unit_of_measurement", "") || "").toLowerCase();
    if(rawUnit === 'w' || (!rawUnit && Math.abs(n) >= 100)) n = n / 1000;
    return n;
  }
  _fmtSignedPowerAuto(idOrIds, digits=2){
    const n = this._powerValueAuto(idOrIds);
    if(n===null) return "—";
    const sign = n > 0 ? '+' : n < 0 ? '−' : '±';
    return `${sign}${Math.abs(n).toFixed(digits)} kW`;
  }
  _fmtAbsPowerAuto(idOrIds, digits=2){

  }
  _themeIsLight(){
    const v = getComputedStyle(this).getPropertyValue('--primary-background-color').trim();
    if(!v) return false;
    let r=255,g=255,b=255;
    if(v.startsWith('#')){
      const h=v.slice(1);
      if(h.length===3){ r=parseInt(h[0]+h[0],16); g=parseInt(h[1]+h[1],16); b=parseInt(h[2]+h[2],16); }
      else if(h.length>=6){ r=parseInt(h.slice(0,2),16); g=parseInt(h.slice(2,4),16); b=parseInt(h.slice(4,6),16); }
    } else {
      const m=v.match(/rgba?\(([^)]+)\)/i);
      if(m){ const p=m[1].split(',').map(s=>parseFloat(s.trim())); [r,g,b]=p; }
    }
    const luminance = (0.2126*r + 0.7152*g + 0.0722*b) / 255;
    return luminance > 0.62;
  }
