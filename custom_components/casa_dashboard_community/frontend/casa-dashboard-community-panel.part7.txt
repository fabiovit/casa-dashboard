")) return "state-on light-tone";
    if(entity.startsWith("climate.")) return "state-on blue-tone";
    return "state-on";
  }

  _overall() {
    const alarm=this._state("alarm_control_panel.home_alarm","unknown");
    const leak=this._on("binary_sensor.bathroom_water_leak");
    const tank=this._on("binary_sensor.garage_dehumidifier_tank_full") || ["full","pieno","water full","tank full"].includes(this._state("sensor.garage_dehumidifier_tank_status","").toLowerCase());
    const open=this._openCount();
    if(alarm==="triggered" || leak || tank) return {level:"bad", icon:"mdi:alert-octagon", title:"Attenzione richiesta", text: alarm==="triggered"?"Allarme in corso":leak?"Perdita acqua rilevata":"Serbatoio deumidificatore pieno"};
    if(open>0) return {level:"warn", icon:"mdi:shield-alert-outline", title:"Casa da controllare", text:`${open} aperture aperte`};
    if(alarm==="disarmed") return {level:"warn", icon:"mdi:shield-off-outline", title:"Casa tranquilla", text:"Allarme disinserito"};
    return {level:"good", icon:"mdi:shield-check-outline", title:"Tutto sotto controllo", text:"Nessuna criticità rilevata"};
  }

  _roomStatus(room) {
    const light=(ids)=>ids.some(id=>this._on(id));
    const active=(ids)=>ids.some(id=>this._on(id));
    switch(room){
      case "esterno": return this._on("input_boolean.pedestrian_gate") ? {level:"warn",text:"Cancellino aperto"}:{level:"good",text:"Accesso chiuso"};
      case "scale": return light(["light.stairs"]) ? {level:"blue",text:"Luce accesa"}:{level:"good",text:"Tutto ok"};
      case "ingresso": {
        if(this._state("alarm_control_panel.home_alarm")==="triggered") return {level:"bad",text:"Allarme in corso"};
        if(this._on("binary_sensor.entry_door")) return {level:"warn",text:"Porta aperta"};
        if(this._on("input_boolean.entry_door_unlock")) return {level:"warn",text:"Porta sbloccata"};
        return {level:"good",text:"Ingresso sicuro"}; }
      case "disimpegno": return active(["binary_sensor.hallway_motion","binary_sensor.hallway_occupancy"]) ? {level:"blue",text:"Presenza rilevata"}:{level:"good",text:"Libero"};
      case "bagno": {
        if(this._on("binary_sensor.bathroom_water_leak")) return {level:"bad",text:"Perdita acqua"};
        if(this._on("binary_sensor.bathroom_window")) return {level:"warn",text:"Finestra aperta"};
        return this._climateActive("fan.bathroom_extractor") ? {level:"blue",text:"Aspiratore attivo"}:{level:"good",text:"Tutto ok"}; }
      case "cucina": {
        if(this._on("binary_sensor.kitchen_door_window")) return {level:"warn",text:"Porta finestra aperta"};
        if(this._climateActive("climate.kitchen")) return {level:"blue",text:"Clima attivo"};
        return light(["light.kitchen_main","light.kitchen_led"]) ? {level:"blue",text:"Luci accese"}:{level:"good",text:"Tutto ok"}; }
      case "camera": {
        if(this._on("binary_sensor.bedroom_door_window")) return {level:"warn",text:"Porta finestra aperta"};
        if(this._climateActive("climate.bedroom")) return {level:"blue",text:"Clima attivo"};
        const v=this._state("vacuum.robot_vacuum",""); if(["cleaning","returning","paused"].includes(v)) return {level:"blue",text:"Robot aspirapolvere attivo"};
        return {level:"good",text:"Tutto ok"}; }
      case "balcone": {
        if(this._on("binary_sensor.weather_rain") || (this._num("sensor.weather_rain_rate",0)>=0.1)) return {level:"warn",text:"Pioggia rilevata"};
        if(this._num("sensor.weather_wind_gust",0)>=24) return {level:"warn",text:"Raffiche forti"};
        if(this._on("switch.balcony_barbecue")) return {level:"blue",text:"Barbecue acceso"};
        return {level:"good",text:"Meteo regolare"}; }
      case "garage": {
        if(this._on("binary_sensor.garage_door_contact")) return {level:"warn",text:"Basculante aperto"};
        if(this._on("binary_sensor.ev_charging_at_home")) return {level:"blue",text:"EV in carica"};
        if(this._on("binary_sensor.garage_dehumidifier_tank_full")) return {level:"bad",text:"Serbatoio pieno"};
        return {level:"good",text:"Tutto ok"}; }
    }
    return {level:"good",text:"Tutto ok"};
  }

  _renderNav(force=false) {
    const nav=this.shadowRoot.querySelector("#nav"); if(!nav) return;
    if(force || !nav.children.length){
      nav.innerHTML=NAV.map(([id,icon,label])=>`<button class="nav-btn ${id===this._active?"active":""}" data-nav="${id}" type="button"><ha-icon icon="${icon}"></ha-icon>${label}</button>`).join("");
    } else {
      nav.querySelectorAll(".nav-btn").forEach(btn=>btn.classList.toggle("active",btn.dataset.nav===this._active));
    }
  }

  }

  }
