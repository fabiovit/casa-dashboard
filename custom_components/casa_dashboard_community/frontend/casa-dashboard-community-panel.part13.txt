    // Preserve the vertical position of the current configurator page before
    // replacing its markup. This prevents add/edit/upload actions from
    // jumping the user back to the top.
    const previousPage=layer.querySelector(".config-page");
    if(previousPage && this._configTab){
      this._configScrollByTab=this._configScrollByTab||{};
      this._configScrollByTab[this._configTab]=previousPage.scrollTop||0;
    }

    layer.innerHTML=this._renderConfigDialog();
    this._applyLanguage(layer);
    if(!this._configDialogOpen) return;

    const restoreConfigScroll=()=>{
      const page=layer.querySelector(".config-page");
      if(page){
        const saved=this._configScrollByTab?.[this._configTab];
        if(Number.isFinite(saved)) page.scrollTop=saved;
      }
    };
    requestAnimationFrame(()=>{ restoreConfigScroll(); requestAnimationFrame(restoreConfigScroll); });
    setTimeout(restoreConfigScroll,60);

    const syncRooms=()=>{
      for(const r of this._roomsDraft){
        const n=layer.querySelector(`[data-room-name="${r.id}"]`); if(n) r.name=n.value.trim()||"Stanza";
        const t=layer.querySelector(`[data-room-type="${r.id}"]`); if(t){ r.type=t.value; if(!r.icon) r.icon=this._roomIcon(r.type); }
        const ic=layer.querySelector(`[data-room-icon="${r.id}"]`); if(ic) r.icon=ic.value.trim()||this._roomIcon(r.type);
        const im=layer.querySelector(`[data-room-image="${r.id}"]`); if(im) r.image=im.value.trim();
        const bm=layer.querySelector(`[data-room-background-mode="${r.id}"]`); if(bm) r.background_mode=bm.value||"auto";
        const bc=layer.querySelector(`[data-room-background-color="${r.id}"]`); if(bc) r.background_color=bc.value||"#101820";
        const ba=layer.querySelector(`[data-room-background-gradient-a="${r.id}"]`); if(ba) r.background_gradient_a=ba.value||"#101820";
        const bb=layer.querySelector(`[data-room-background-gradient-b="${r.id}"]`); if(bb) r.background_gradient_b=bb.value||"#183040";
        const bang=layer.querySelector(`[data-room-background-gradient-angle="${r.id}"]`); if(bang) r.background_gradient_angle=Number(bang.value||135);

        // Modules are part of the room itself. Capture every visible checkbox

      this._imageUploadNotice=`Elaborazione foto: ${file.name||"immagine"}…`;
      const setImage=(url)=>{
        if(entity){
          this._setRoomEntityImage(roomId,entity,url);
        }else{
          const room=this._roomsDraft.find(r=>r.id===roomId);
          if(room) room.image=url;
        }
      };

      try{
        const uploaded=await this._uploadDashboardImage(file,(inline)=>{
          // Immediate result: persist in draft and show preview now.
          setImage(inline);
          this._imageUploadNotice="Foto acquisita. Ottimizzazione del salvataggio in corso…";
          this._renderConfigLayer();
        });

        if(uploaded?.url) setImage(uploaded.url);
        this._imageUploadNotice=uploaded?.mode==="file"
          ? "Foto caricata e verificata."
          : "Foto acquisita e pronta per il salvataggio.";
        this._renderConfigLayer();
      }catch(err){
        this._imageUploadNotice=`Errore foto: ${err?.message||String(err)}`;
        this._renderConfigLayer();
        alert(this._imageUploadNotice);
      }
    };

    layer.querySelectorAll("[data-room-photo-file]").forEach(input=>{
      input.addEventListener("change",()=>handleNativePhoto(input,input.dataset.roomPhotoFile));
    });
    layer.querySelectorAll("[data-overview-photo-file]").forEach(input=>{
      input.addEventListener("change",async()=>{
        const file=input.files?.[0]; if(!file) return;
        const key=input.dataset.overviewPhotoFile;
        this._imageUploadNotice=`Elaborazione foto: ${file.name||"immagine"}…`;
        try{
          const uploaded=await this._uploadDashboardImage(file,(inline)=>{
            this._overviewDraft[key]=inline;
            this._imageUploadNotice="Foto acquisita. Ottimizzazione del salvataggio in corso…";
            this._renderConfigLayer();
          });
          if(uploaded?.url) this._overviewDraft[key]=uploaded.url;
          this._imageUploadNotice=uploaded?.mode==="file"?"Foto caricata e verificata.":"Foto acquisita e pronta per il salvataggio.";
          this._renderConfigLayer();
        }catch(err){
          this._imageUploadNotice=`Errore foto: ${err?.message||String(err)}`;
          alert(this._imageUploadNotice);
          this._renderConfigLayer();
        }
      });
    });
    layer.querySelectorAll("[data-entity-photo-file-room]").forEach(input=>{
      input.addEventListener("change",()=>handleNativePhoto(input,input.dataset.entityPhotoFileRoom,input.dataset.entityPhotoFile));
    });

    layer.addEventListener("click",async ev=>{
      const clearRoom=ev.target.closest?.("[data-clear-room-image]");
      const clearEntity=ev.target.closest?.("[data-clear-entity-image-room]");
      const clearOverview=ev.target.closest?.("[data-clear-overview-image]");
      if(clearOverview){
        const key=clearOverview.dataset.clearOverviewImage;
        if(key) delete this._overviewDraft[key];
        this._renderConfigLayer(); return;
      }
      if(clearRoom){
        ev.preventDefault();
        const room=this._roomsDraft.find(r=>r.id===clearRoom.dataset.clearRoomImage);
        if(room){room.image="";this._renderConfigLayer();} return;
      }
      if(clearEntity){
        ev.preventDefault();
        this._setRoomEntityImage(clearEntity.dataset.clearEntityImageRoom,clearEntity.dataset.clearEntityImage,"");
        this._renderConfigLayer(); return;
      }
      const add=ev.target.closest?.("[data-room-entity-add]");
      if(add){
        ev.preventDefault();
        const room=this._roomsDraft.find(r=>r.id===add.dataset.roomEntityAdd), entity=add.dataset.entity;
        if(room && entity && this._hass?.states?.[entity] && !(room.entities||[]).includes(entity)){
          room.entities=[...(room.entities||[]),entity]; this._renderConfigLayer();
        }
        return;
      }
      const sortType=ev.target.closest?.("[data-room-sort-type]");
      if(sortType){
        const room=this._roomsDraft.find(r=>r.id===sortType.dataset.roomSortType);
        if(room){
          room.entity_order=[...(room.entities||[])].sort((a,b)=>{
            const pa=this._roomEntityTypePriority(a),pb=this._roomEntityTypePriority(b);
            return pa!==pb?pa-pb:this._roomEntityDisplay(a,room).localeCompare(this._roomEntityDisplay(b,room),this._language==="en"?"en":"it");
          });
          this._renderConfigLayer();
        }
        return;
      }
      const moveUp=ev.target.closest?.("[data-room-entity-up]"),moveDown=ev.target.closest?.("[data-room-entity-down]");
      if(moveUp||moveDown){
        const btn=moveUp||moveDown,roomId=moveUp?btn.dataset.roomEntityUp:btn.dataset.roomEntityDown,entity=btn.dataset.entity;
        const room=this._roomsDraft.find(r=>r.id===roomId);
        if(room){
          const arr=this._orderedRoomEntities(room),i=arr.indexOf(entity),j=i+(moveUp?-1:1);
          if(i>=0&&j>=0&&j<arr.length)[arr[i],arr[j]]=[arr[j],arr[i]];
          room.entity_order=arr; this._renderConfigLayer();
        }
        return;
      }
      const rem=ev.target.closest?.("[data-room-entity-remove]");
      if(rem){
        ev.preventDefault();
        const room=this._roomsDraft.find(r=>r.id===rem.dataset.roomEntityRemove);
        if(room){
          room.entities=(room.entities||[]).filter(e=>e!==rem.dataset.entity);
          if(room.entity_labels)delete room.entity_labels[rem.dataset.entity];
          if(room.entity_images)delete room.entity_images[rem.dataset.entity];
          this._renderConfigLayer();
        }
      }
    });
  }
