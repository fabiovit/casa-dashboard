     <div class="hallway-frame left"></div>
          <div class="hallway-frame right"></div>
          <div class="hallway-cone left"></div>
          <div class="hallway-cone right"></div>
          <div class="hallway-person ${motionOn?'active':''}"><ha-icon icon="mdi:walk"></ha-icon></div>
        </div>
        <div class="hallway-hero-meta">
          <div><span>MOVIMENTO</span><strong>${motionOn?"Presente":"Assente"}</strong><small>${motion?display(motion):"Sensore passaggio"}</small></div>
          <div class="hallway-meta-divider"></div>
          <div><span>OCCUPAZIONE</span><strong>${occupied?"Occupato":"Libero"}</strong><small>Presenza nel passaggio</small></div>
        </div>
      </div>
    </section>`;

    const lightCards=lights.map(e=>{
      const hay=`${e} ${display(e)}`.toLowerCase();
      const icon=/led|strip|striscia/.test(hay)?"mdi:led-strip-variant":/lamp/.test(hay)?"mdi:lamp":"mdi:ceiling-light";
      return this._lightVisual(e,display(e),icon,room.name,6);
    }).join("");

    const applianceCards=switches.map(e=>this._hallwayApplianceVisual(e,display(e),6)).join("");

    const sensorCards=[
      ...sensors.map(e=>{
        const dc=String(this._attr(e,"device_class","")).toLowerCase();
        const name=display(e);
        const unit=this._attr(e,"unit_of_measurement","")||"";
        if(dc==="temperature") return this._environmentVisual(e,name,"temperature",4,room.name);
        if(dc==="humidity") return this._environmentVisual(e,name,"humidity",4,room.name);
        if(dc==="illuminance" || /lux/i.test(String(unit))) return this._environmentVisual(e,name,"lux",4,room.name);
        return this._metric(this._v1OriginalIcon(e,room.type,name),name,this._state(e),unit,4,e);
      }),
      ...binaries.map(e=>{
        const dc=String(this._attr(e,"device_class","")).toLowerCase();
        const name=display(e);
        if(["motion","occupancy","presence","vibration"].includes(dc)){
          const type=dc==="motion"?"motion":dc==="vibration"?"vibration":"occupancy";
          return this._activityVisual(e,name,type,4,room.name);
        }
        if(["door","window","opening","garage_door"].includes(dc)){
          const kind=dc==="window"?"window":dc==="garage_door"?"garage":"door";
          return this._openingVisual(e,name,kind,4,room.name);
        }
        return this._metric(this._v1OriginalIcon(e,room.type,name),name,this._on(e)?"Attivo":"Inattivo","",4,e);
      })
    ].join("");

    return `<div class="page hallway-dynamic-page">
      ${hero}
      ${lightCards?this._section("Luci"):""}
      <div class="grid">${lightCards}</div>
      ${applianceCards?this._section("Dispositivi ed elettrodomestici"):""}
      <div class="grid">${applianceCards}</div>
      ${sensorCards?this._section("Controllo e sensori"):""}
      <div class="grid">${sensorCards}</div>
      ${this._renderRoomModules(room)}
    </div>`;
  }

  _dynamicRoom(room){
    const entities=this._orderedRoomEntities(room).filter(e=>this._hass?.states?.[e]);
    const display=e=>this._roomEntityDisplay(e,room);
    const byDomain=d=>entities.filter(e=>e.startsWith(d+"."));
    const lights=byDomain("light");
    const climates=byDomain("climate");
    const covers=byDomain("cover");
    const sensors=[...byDomain("sensor"),...byDomain("binary_sensor")];
    const switches=[...byDomain("switch"),...byDomain("fan"),...byDomain("humidifier"),...byDomain("input_boolean")];
    const locks=byDomain("lock");
    const alarms=byDomain("alarm_control_panel");
    const scripts=byDomain("script");
    const buttons=byDomain("button");
    const media=byDomain("media_player");
    const vacuums=byDomain("vacuum");
    const mowers=byDomain("lawn_mower");

    const temp=sensors.find(e=>{
      const dc=String(this._attr(e,"device_class","")).toLowerCase();
      const name=String(display(e));
      return dc==="temperature" || /temperatura|temperature/i.test(name);
    });
    const hum=sensors.find(e=>{
      const dc=String(this._attr(e,"device_class","")).toLowerCase();
      const name=String(display(e));
      return dc==="humidity" || /umidità|humidity/i.test(name);
    });
    const opening=sensors.find(e=>{
      const dc=String(this._attr(e,"device_class","")).toLowerCase();
      return ["door","window","opening","garage_door"].includes(dc);
    });
    const motion=sensors.find(e=>{
      const dc=String(this._attr(e,"device_class","")).toLowerCase();
      return ["motion","occupancy","vibration"].includes(dc);
    });
    const lockLikeBoolean=switches.find(e=>{
      if(!e.startsWith("input_boolean.")) return false;
      const hay=`${e} ${display(e)}`.toLowerCase();
      return /serratura|sblocc|porta.*lock|door.*lock|unlock/.test(hay);
    })||null;
    const isEntryRoom=String(room.type||"").toLowerCase()==="ingresso"
      || /ingresso|entrance|entry/i.test(String(room.name||""));
    const mappedEntryLock=isEntryRoom && this._configured("input_boolean.entry_door_unlock")
      ? this._eid("input_boolean.entry_door_unlock")
      : null;
    const mappedEntryAlarm=isEntryRoom && this._configured("alarm_control_panel.home_alarm")
      ? this._eid("alarm_control_panel.home_alarm")
      : null;
    const roomLock=locks[0]||lockLikeBoolean||mappedEntryLock||null;
    const roomAlarm=alarms[0]||mappedEntryAlarm||null;
    const roomLockIsBoolean=!!roomLock && roomLock.startsWith("input_boolean.");
    const rawLockState=roomLock?this._state(roomLock,"unknown"):"";
    const lockState=roomLockIsBoolean
      ? (this._on(roomLock)?"unlocked":"locked")
      : rawLockState;
    const alarmState=roomAlarm?this._state(roomAlarm,"unknown"):"";
    const alarmLabel=state=>{
      const map={
        disarmed:"Non armato",
        armed_home:"Armato · notte",
        armed_away:"Armato · fuori casa",
        armed_night:"Armato · notte",
        armed_vacation:"Armato · vacanza",
        arming:"Inserimento…",
        pending:"In attesa…",
        triggered:"ALLARME"
      };
      return map[state]||state||"—";
    };

    const lightOn=lights.filter(e=>this._on(e)).length;
    const climateOn=climates.filter(e=>this._climateActive(e)).length;
    const switchOn=switches.filter(e=>this._on(e)).length;
    const openingOn=opening&&this._on(opening);
    const motionOn=motion&&this._on(motion);
    const securityActive=(roomAlarm && alarmState==="triggered")?1:0;
    const activeCount=lightOn+climateOn+switchOn+(openingOn?1:0)+(motionOn?1:0)+securityActive;

    const firstDevice=[...switches,...media,...scripts,...buttons][0]||null;
    const metrics=[
      opening?{
        label:"Apertura",
        value:this._on(opening)?"Aperta":"Chiusa",
        small:display(opening),
        show:true
      }:null,
      roomLock?{
        label:"Porta",
        value:lockState==="locked"?"Bloccata":lockState==="unlocked"?"Sbloccata":this._esc(lockState),
        small:display(roomLock),
        show:true
      }:null,
      roomAlarm?{
        label:"Allarme",
        value:alarmLabel(alarmState),
        small:display(roomAlarm),
        show:true
      }:null,
      temp?{
        label:"Comfort",
        value:`${this._state(temp)}${this._attr(temp,"unit_of_measurement","")||""}`,
        small:hum?`Umidità ${this._state(hum)}${this._attr(hum,"unit_of_measurement","")||""}`:"",
        show:true
      }:null,
      climates[0]?{
        label:"Clima",
        value:this._climateActive(climates[0])?"Attivo":"Spento",
        small:display(climates[0]),
        show:true
      }:null,
      lights.length?{
        label:"Luci",
        value:lightOn?`${lightOn} accese`:"Spente",
        small:lights.length>1?`${lights.length} punti luce`:(display(lights[0])),
        show:true
      }:null,
      firstDevice?{
        label:"Dispositivo",
        value:firstDevice.startsWith("script.")||firstDevice.startsWith("button.")?"Disponibile":this._localizedState(firstDevice,this._state(firstDevice,"Pronto")),
        small:display(firstDevice),
        show:true
      }:null
    ].filter(Boolean).slice(0,3);

    const roomLabel=this._roomTypes().find(x=>x[0]===room.type)?.[1]||"Ambiente";
    const hero=this._hero(
      room.name,
      this._uiTheme==="v1"
        ? `${roomLabel} · comfort, luci ed elettrodomestici`
        : `${roomLabel} · comfort, luci e dispositivi`,
      {level:activeCount?"warn":"good",title:activeCount?`${activeCount} stati attivi`:"Tutto tranquillo",text:""},
      metrics,
      room.type||"generico",
      room
    );

    const securityAlert=(opening && roomLock && !this._on(opening) && lockState==="unlocked")
      ? this._alert("warn","mdi:lock-open-alert-outline","Porta chiusa ma sbloccata","La serratura risulta aperta anche se il contatto porta è chiuso.")
      : "";

    const lockPhoto=roomLock?this._activeRoomEntityImage(roomLock):"";
    const lockStateLabel=roomLock?this._localizedState(roomLock,lockState):"—";
    const lockCard=roomLock?`<div class="community-security-card door-security smart-lock-card span-6${this._entitySizeClass(roomLock)} ${lockPhoto?"has-custom-photo":""}">
      ${lockPhoto
        ? `<div class="security-lock-photo"><img src="${this._esc(lockPhoto)}" alt="${this._esc(display(roomLock))}" loading="lazy"></div>`
        : `<div class="security-door-visual">
            <div class="security-door-frame"><div class="security-door-leaf"><i></i></div></div>
            <span class="security-badge ${lockState==="locked"?"good":"warn"}">${this._esc(lockStateLabel).toUpperCase()}</span>
          </div>`}
      <div class="security-card-copy">
        <span>${this._esc(display(roomLock))}</span>
        <strong>${opening?`${this._on(opening)?"Aperta":"Chiusa"} · `:""}${this._esc(lockStateLabel)}</strong>
        <small>${opening?"Stato porta + serratura":"Serratura smart"}</small>
        <div class="security-actions">
          <button class="security-btn primary" data-lock-action="${lockState==="locked"?"unlock":"lock"}" data-lock-entity="${roomLock}" type="button">${lockState==="locked"?"Sblocca porta":"Blocca porta"}</button>
          <button class="security-btn" data-more="${roomLock}" type="button">Dettagli</button>
        </div>
      </div>
    </div>`:"";

    const alarmCard=roomAlarm?`<div class="community-security-card alarm-security ${roomLock?"span-6":"span-12"} ${alarmState==="triggered"?"triggered":""}">
      <div class="alarm-card-main">
        <div class="alarm-title-row">
          <div class="alarm-shield"><ha-icon icon="${alarmState==="triggered"?"mdi:alarm-light":"mdi:shield-home"}"></ha-icon></div>
          <div><span>PANNELLO ALLARME</span><strong>${alarmLabel(alarmState)}</strong><small>${this._esc(display(roomAlarm))}</small></div>
        </div>
        <div class="alarm-quick-actions">
          <button class="${["armed_home","armed_night"].includes(alarmState)?"active":""}" data-alarm-service="alarm_arm_home" data-alarm-entity="${roomAlarm}" type="button">NOTTE</button>
          <button class="${alarmState==="armed_away"?"active":""}" data-alarm-service="alarm_arm_away" data-alarm-entity="${roomAlarm}" type="button">FUORI CASA</button>
          <button class="${alarmState==="disarmed"?"active":""}" data-alarm-service="alarm_disarm" data-alarm-entity="${roomAlarm}" type="button">OFF</button>
        </div>
      </div>
      <div class="alarm-keypad">
        <span>ALLARME</span><strong>${alarmLabel(alarmState)}</strong>
        <div class="keypad-grid">${["1","2","3","4","5","6","7","8","9"].map(n=>`<i>${n}</i>`).join("")}<i><ha-icon icon="mdi:lock-outline"></ha-icon></i><i>0</i><i><ha-icon icon="mdi:shield-check-outline"></ha-icon></i></div>
      </div>
    </div>`:"";

    const genericSwitches=switches.filter(e=>e!==roomLock && e!==mappedEntryLock);
    const controlCards=[
      ...lights.map(e=>this._control(
        e,
        display(e),
        this._v1OriginalIcon(e,room.type,display(e)),
        this._on(e)?"Accesa":"Spenta",
        "toggle",6
      )),
      ...genericSwitches.map(e=>this._isRoomFanDevice(e,room,display(e))
        ? this._roomFanVisual(e,display(e),6,"Ventilazione")
        : this._control(
            e,
            display(e),
            this._v1OriginalIcon(e,room.type,display(e)),
            this._on(e)?"Acceso":"Spento",
            "toggle",6
          )),
      ...media.map(e=>this._universalDeviceVisual(
        e,display(e),"generic",6,"",
        {kind:"more",icon:this._v1OriginalIcon(e,room.type,display(e)),status:this._localizedState(e,this._state(e,"—")),actionText:"DETTAGLI"}
      )),
      ...vacuums.map(e=>this._vacuumRoomCard(e,display(e),6)),
      ...mowers.map(e=>this._mowerRoomCard(e,display(e),6)),
      ...scripts.map(e=>`<div class="command-strip span-6"><div class="command-symbol"><ha-icon icon="${this._v1OriginalIcon(e,room.type,display(e))}"></ha-icon></div><div class="command-copy"><div class="command-title">${this._esc(display(e))}</div><div class="command-sub">Azione</div></div><button class="command-action" data-script="${e}" type="button"><span>ESEGUI</span><i></i></button></div>`),
      ...buttons.map(e=>`<div class="command-strip span-6"><div class="command-symbol"><ha-icon icon="${this._v1OriginalIcon(e,room.type,display(e))}"></ha-icon></div><div class="command-copy clickable" data-more="${e}"><div class="command-title">${this._esc(display(e))}</div><div class="command-sub">Comando</div></div></div>`)
    ].join("");

    const coverCards=covers.map(e=>{
      const label=display(e);
      if(this._isAwningCover(e,room,label)) return this._awningVisual(e,label,6);
      return this._uiTheme==="v1"
        ? `<div class="command-strip span-6"><div class="command-symbol"><ha-icon icon="${this._v1OriginalIcon(e,room.type,label)}"></ha-icon></div><div class="command-copy clickable" data-more="${e}"><div class="command-title">${this._esc(label)}</div><div class="command-sub">${this._esc(this._state(e,"—"))}</div></div><button class="command-action" data-cover-toggle="${e}" data-inverted="0" type="button"><span>GESTISCI</span><i></i></button></div>`
        : this._coverControl(e,label,false,6);
    }).join("");

    const climateCards=climates.map(e=>this._climateBlock(
      e,"","",
      this._hass.states[e]?.attributes?.friendly_name||room.name,""
    )).join("");

    const sensorPriority=(e)=>{
      const obj=this._hass.states[e];
      const dc=String(obj?.attributes?.device_class||"").toLowerCase();
      const unit=String(obj?.attributes?.unit_of_measurement||"").toLowerCase();
      const hay=`${e} ${display(e)}`.toLowerCase();

      // Desired visual order:
      // 1. doors
      // 2. windows
      // 3. illuminance
      // 4. temperature
      // 5. motion / occupancy
      // 6. everything else
      if(dc==="door" || dc==="garage_door" || /porta|door/.test(hay)) return 10;
      if(dc==="window" || /finestra|window/.test(hay)) return 20;
      if(dc==="illuminance" || unit==="lx" || /lux|luminos|illumin/.test(hay)) return 30;
      if(dc==="temperature" || /temperatura|temperature/.test(hay)) return 40;
      if(["motion","occupancy","presence"].includes(dc) || /movimento|motion|occupazione|presence/.test(hay)) return 50;
      return 100;
    };

    const sortedSensors=[...sensors].sort((a,b)=>{
      const pa=sensorPriority(a), pb=sensorPriority(b);
      if(pa!==pb) return pa-pb;
      return display(a).localeCompare(display(b),this._language==="en"?"en":"it",{sensitivity:"base"});
    });

    const sensorCards=sortedSensors.map(e=>{
      const obj=this._hass.states[e];
      const name=display(e);
      const dc=String(obj?.attributes?.device_class||"").toLowerCase();
      const unit=obj?.attributes?.unit_of_measurement||"";

      // Match the visual language of the original Casa Dashboard.
      if(["motion"].includes(dc)) return this._activityVisual(e,name,"motion",4,room.name);
      if(["occupancy","presence"].includes(dc)) return this._activityVisual(e,name,"occupancy",4,room.name);
      if(["vibration"].includes(dc)) return this._activityVisual(e,name,"vibration",4,room.name);
      if(["door","window","opening","garage_door"].includes(dc) && typeof this._openingVisual==="function"){
        const kind=dc==="window"?"window":dc==="garage_door"?"garage":"door";
        return this._openingVisual(e,name,kind,4,room.name);
      }
      if(dc==="temperature") return this._environmentVisual(e,name,"temperature",4,room.name);
      if(dc==="humidity") return this._environmentVisual(e,name,"humidity",4,room.name);
      if(dc==="illuminance" || /lux/i.test(String(unit))) return this._environmentVisual(e,name,"lux",4,room.name);

      const icon=this._v1OriginalIcon(e,room.type,name);
      const raw=this._state(e,"—");
      const semantic=(dc==="door"||dc==="window"||dc==="opening")
        ? (this._on(e)?"Aperta":"Chiusa")
        : (Number.isFinite(Number(raw)) ? raw : this._localizedState(e,raw));
      return this._metric(icon,name,semantic,unit,4,e);
    }).join("");

    const hasControls=!!(controlCards||coverCards||climateCards);
    const hasSecurity=!!(lockCard||alarmCard);
    return `<div class="page dynamic-room-page">
      ${hero}
      ${hasControls?this._section(this._uiTheme==="v1"?"Luci, tende ed elettrodomestici":"Luci, tende e dispositivi"):""}
      <div class="grid">${controlCards}${coverCards}${climateCards}</div>
      ${securityAlert?`<div class="community-security-alert">${securityAlert}</div>`:""}
      ${hasSecurity?this._section("Sicurezza e accesso"):""}
      ${hasSecurity?`<div class="community-security-grid entity-aware ${lockCard&&alarmCard?"two":"single"}">${lockCard}${alarmCard}</div>`:""}
      ${sensorCards?this._section(`Sensori ${this._esc(room.name)}`,"Stato dell'ambiente"):""}
      <div class="grid">${sensorCards}</div>
      ${this._renderRoomModules(room)}
    </div>`;
  }

  _entityOptions() {
    if(!this._hass?.states) return [];
    return Object.entries(this._hass.states).map(([id,state])=>{
      const name=String(state?.attributes?.friendly_name || "").trim();
      return {id,name,label:name ? `${name} — ${id}` : id};
    }).sort((a,b)=>(a.name || a.id).localeCompare(b.name || b.id, "it", {sensitivity:"base"}));
  }
  _entityIdFromFriendlyName(value) {
    const raw=String(value || "").trim();
    if(!raw || raw.includes(".")) return raw;
    const normalized=raw.toLocaleLowerCase("it");
    const matches=this._entityOptions().filter(item=>item.name && item.name.toLocaleLowerCase("it")===normalized);
    return matches.length===1 ? matches[0].id : raw;
  }
