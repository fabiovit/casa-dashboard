class CasaDashboardPanel extends HTMLElement {
  constructor() {
    super();
    this.attachShadow({ mode: "open" });
    this._hass = null;
    this._panel = null;
    this._active = "panoramica";
    this._rendered = false;
    this._optimistic = new Map();
    this._renderTimer = null;
    this._pointerBusy = false;
    this._renderPending = false;
    this._entityMap = {};
    this._entityConfigLoaded = false;
    this._entityConfigLoading = false;
    this._configDialogOpen = false;
    this._configFilter = "";
    this._configDraft = {};
    this._rooms = [];
    this._roomsDraft = [];
    this._lastStateFingerprint = "";
    try {
      const saved=localStorage.getItem("casa_dashboard_community_style");
      const migrated=saved==="community"?"v1":saved==="casa4354"?"v2":saved;
      this._uiTheme=["v1","v2"].includes(migrated)?migrated:"v1";
    } catch(e) { this._uiTheme="v1"; }
    try {
      const savedLang=localStorage.getItem("casa_dashboard_community_language");
      this._language=["it","en"].includes(savedLang)?savedLang:((navigator.language||"it").toLowerCase().startsWith("en")?"en":"it");
    } catch(e) { this._language="it"; }
  }

  set panel(value) { this._panel = value; }
  set hass(value) {
    this._hass = value;
    this._syncOptimistic();
    if (!this._rendered) this._renderShell();
    this._loadEntityConfig();
    // Home Assistant publishes many hass updates that are unrelated to this dashboard.
    // Rebuilding the whole page on every update replaces hovered cards in the DOM and
    // causes visible flicker. Once the entity map is loaded, render only when one of
    // the configured entities has actually changed.
    if (!this._entityConfigLoaded || this._stateFingerprint() !== this._lastStateFingerprint) {
      this._scheduleRender();
    }
  }

  _upgradeProperty(prop) {
    // Home Assistant may assign panel/hass before this custom element is defined.
    // In that case the value becomes an own property and shadows our setter after upgrade.
    // Re-apply it through the prototype setter so the dashboard receives HA state correctly.
    if (Object.prototype.hasOwnProperty.call(this, prop)) {
      const value = this[prop];
      delete this[prop];
      this[prop] = value;
    }
  }

  connectedCallback() {
    this._upgradeProperty("panel");
    this._upgradeProperty("hass");
    if (!this._rendered) this._renderShell();
    this._loadEntityConfig();
    this._renderPage();
  }

  _renderShell() {
    this._rendered = true;
    this.shadowRoot.innerHTML = `
      <style>${this._styles()}</style>
      <div class="app-shell ui-${this._uiTheme}">

              </div>
            </div>
          <div class="language-switch" aria-label="Language"><button type="button" data-language="it">IT</button><button type="button" data-language="en">EN</button></div>
          </div>
          <div class="nav-scroller"><nav class="nav" id="nav"></nav></div>
        </header>

    `;
    this.shadowRoot.addEventListener("click", (ev) => this._handleClick(ev));
    this.shadowRoot.querySelectorAll("[data-language]").forEach((button)=>{
      button.addEventListener("click",(ev)=>{
        ev.preventDefault();
        ev.stopPropagation();
        this._setLanguage(button.dataset.language);
      });
    });
    this._applyLanguage();
  }
