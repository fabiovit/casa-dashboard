  _roomCard(room) {
    if(!this._roomConfigured(room)) return "";
    const m=ROOM_META[room], s=this._roomStatus(room); let tags=[];
    if(room==="bagno") {
      if(this._configured("sensor.bathroom_temperature")) tags.push(`🌡️ ${this._fmt("sensor.bathroom_temperature",1,"°C")}`);
      if(this._configured("sensor.bathroom_humidity")) tags.push(`💧 ${this._fmt("sensor.bathroom_humidity",0,"%")}`);
    }
    if(room==="cucina") {
      if(this._configured("sensor.kitchen_temperature")) tags.push(`🌡️ ${this._fmt("sensor.kitchen_temperature",1,"°C")}`);
      if(this._configured("sensor.kitchen_humidity")) tags.push(`💧 ${this._fmt("sensor.kitchen_humidity",0,"%")}`);
    }
    if(room==="camera") {
      if(this._configured("sensor.bedroom_temperature")) tags.push(`🌡️ ${this._fmt("sensor.bedroom_temperature",1,"°C")}`);
      if(this._configured("sensor.bedroom_humidity")) tags.push(`💧 ${this._fmt("sensor.bedroom_humidity",0,"%")}`);
    }
    if(room==="balcone") {
      if(this._configured("sensor.weather_outdoor_temperature")) tags.push(`🌡️ ${this._fmt("sensor.weather_outdoor_temperature",1,"°C")}`);
      if(this._configured("sensor.weather_wind_gust")) tags.push(`💨 ${this._fmt("sensor.weather_wind_gust",0,"km/h")}`);
    }
    if(room==="garage") {
      if(this._configured("sensor.garage_temperature")) tags.push(`🌡️ ${this._fmt("sensor.garage_temperature",1,"°C")}`);
      if(this._configured("binary_sensor.ev_charging_at_home") || this._configured("sensor.wallbox_evse_status")) tags.push(this._on("binary_sensor.ev_charging_at_home")?"🔌 In carica":"🚗 Wallbox pronta");
    }
    if(room==="ingresso") {
      if(this._configured("alarm_control_panel.home_alarm")) tags.push(this._state("alarm_control_panel.home_alarm") === "disarmed"?"🔓 Allarme OFF":"🛡️ Allarme ON");
      if(this._configured("light.entry_main")) tags.push(this._on("light.entry_main")?"💡 Luce ON":"💡 Luce OFF");
    }
    return `<article class="card room-card premium-room clickable span-4" data-nav="${room}">
      <div class="room-heading">
        <div class="room-orb ${s.level}"><ha-icon icon="${m.icon}"></ha-icon></div>
        <div class="room-arrow"><ha-icon icon="mdi:arrow-top-right"></ha-icon></div>
      </div>
      <div class="room-copy">
        <div class="room-eyebrow">${m.subtitle}</div>
        <h3>${m.title}</h3>
        <div class="room-status-line ${s.level}">${this._esc(s.text)}</div>
      </div>
      <div class="room-pills">
        ${tags.map(t=>`<span class="room-pill">${t}</span>`).join("")}
      </div>
    </article>`;
  }
