  _overviewMetric(entity, kind="generic"){
    const obj=this._hass?.states?.[entity];
    if(!obj) return {value:"—",unit:"",sub:""};
    const raw=this._state(entity,"—");
    const unit=String(obj.attributes?.unit_of_measurement||"").trim();
    const dc=String(obj.attributes?.device_class||"").toLowerCase();
    const n=parseFloat(raw);
    const finite=Number.isFinite(n);

    if(kind==="power" || dc==="power" || /(^|\s)(w|kw)$/i.test(unit)){
      if(!finite) return {value:raw,unit,sub:""};
      const lower=unit.toLowerCase();
      const kw=lower==="w" ? n/1000 : lower==="kw" ? n : (Math.abs(n)>=100 ? n/1000 : n);
      return {value:new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{minimumFractionDigits:2,maximumFractionDigits:2}).format(kw),unit:"kW",sub:""};
    }
    if(dc==="energy" || /kwh|wh/i.test(unit)){
      if(!finite) return {value:raw,unit,sub:""};
      let value=n, outUnit=unit||"kWh";
      if(unit.toLowerCase()==="wh"){ value=n/1000; outUnit="kWh"; }
      return {value:new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{maximumFractionDigits:2}).format(value),unit:outUnit,sub:""};
    }
    if(dc==="battery" || unit==="%"){
      return {value:finite?new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{maximumFractionDigits:0}).format(n):raw,unit:unit||"%",sub:""};
    }
    if(dc==="temperature" || /°c|°f/i.test(unit)){
      return {value:finite?new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{minimumFractionDigits:1,maximumFractionDigits:1}).format(n):raw,unit,sub:""};
    }
    if(entity.startsWith("binary_sensor.")){
      const on=raw==="on";
      return {value:on?(this._language==="en"?"Active":"Attivo"):(this._language==="en"?"Inactive":"Inattivo"),unit:"",sub:""};
    }
    if(finite){
      return {value:new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{maximumFractionDigits:2}).format(n),unit,sub:""};
    }
    return {value:raw,unit,sub:""};
  }

  _energyPowerKW(logicalKey){
    const entity=this._eid(logicalKey);
    const obj=entity?this._hass?.states?.[entity]:null;
    if(!obj) return null;
    const raw=parseFloat(obj.state);
    if(!Number.isFinite(raw)) return null;
    const unit=String(obj.attributes?.unit_of_measurement||"").trim().toLowerCase();
    if(unit==="w") return raw/1000;
    if(unit==="mw") return raw*1000;
    return raw;
  }

    return Number.isFinite(n)?n:null;
  }
  _energyFmtKW(value,abs=false){
    if(value===null || !Number.isFinite(value)) return "—";
    const n=abs?Math.abs(value):value;
    return `${new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{minimumFractionDigits:2,maximumFractionDigits:2}).format(n)} kW`;
  }
  _energyFmtLogical(logicalKey,digits=2,fallbackUnit=""){
    const entity=this._eid(logicalKey);
    if(!entity || !this._hass?.states?.[entity]) return "—";
    const obj=this._hass.states[entity], n=parseFloat(obj.state);
    const unit=String(obj.attributes?.unit_of_measurement||fallbackUnit||"").trim();
    if(!Number.isFinite(n)) return this._esc(obj.state);
    return `${new Intl.NumberFormat(this._language==="en"?"en-GB":"it-IT",{minimumFractionDigits:digits,maximumFractionDigits:digits}).format(n)}${unit?` ${unit}`:""}`;
  }
