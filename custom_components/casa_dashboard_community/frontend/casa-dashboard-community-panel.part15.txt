  _safeColor(value,fallback="#101820"){
    const v=String(value||"").trim();
    return /^#[0-9a-f]{6}$/i.test(v)?v:fallback;
  }
  _safeBackgroundCss(mode,opts={}){
    const m=String(mode||"auto").toLowerCase();
    if(m==="image"){
      const src=String(opts.image||"").trim();
      if(!src || !(src.startsWith("/local/") || /^https?:\/\//i.test(src) || src.startsWith("data:image/"))) return "";
      return `url("${src.replace(/["\\]/g,"")}") center / cover fixed no-repeat`;
    }
    if(m==="gradient"){
      const a=this._safeColor(opts.gradientA,"#101820");
      const b=this._safeColor(opts.gradientB,"#183040");
      const angle=Math.max(0,Math.min(360,Number(opts.angle||135)));
      return `linear-gradient(${angle}deg,${a},${b})`;
    }
    if(m==="color") return this._safeColor(opts.color,"#101820");
    return "";
  }
  _applyCustomBackground(room=null){
    const shell=this.shadowRoot?.querySelector(".app-shell");
    const host=this.shadowRoot?.querySelector("#content");
    if(!shell || !host) return;

    const gm=this._overviewText("dashboard_background_mode","auto");
    const globalBg=this._safeBackgroundCss(gm,{
      image:this._overviewText("dashboard_background_image",""),
      color:this._overviewText("dashboard_background_color","#101820"),
      gradientA:this._overviewText("dashboard_background_gradient_a","#101820"),
      gradientB:this._overviewText("dashboard_background_gradient_b","#183040"),
      angle:Number(this._overviewText("dashboard_background_gradient_angle","135"))
    });

    const rm=room?.background_mode||"auto";
    const roomBg=this._safeBackgroundCss(rm,{
      image:room?.image||"",
      color:room?.background_color||"#101820",
      gradientA:room?.background_gradient_a||"#101820",
      gradientB:room?.background_gradient_b||"#183040",
      angle:Number(room?.background_gradient_angle||135)
    });

    const mode=roomBg?rm:gm;
    const activeBg=roomBg||globalBg;
    shell.style.background=activeBg||"";
    shell.style.backgroundAttachment=activeBg && mode==="image" ? "fixed" : "";

  }

  _metric(icon,label,value,foot="",span=3,entity="",tone="") {
