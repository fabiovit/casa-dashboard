  _handleClick(ev) {
    const target=ev.target.closest?.("[data-action],[data-nav],[data-toggle],[data-more],[data-cover-toggle],[data-climate],[data-climate-temp],[data-vacuum],[data-alarm],[data-humidity-delta],[data-script]"); if(!target) return;
    if(target.dataset.action==="ha-menu") { this.dispatchEvent(new CustomEvent("hass-toggle-menu",{bubbles:true,composed:true})); return; }
    if(target.dataset.nav){
      this._active=target.dataset.nav; this._renderNav(false); this._renderPage(false);
      requestAnimationFrame(()=>this.shadowRoot.querySelector(`.nav-btn[data-nav="${this._active}"]`)?.scrollIntoView({behavior:"smooth",inline:"center",block:"nearest"}));
      window.scrollTo({top:0,behavior:"smooth"}); return;
    }
    if(target.dataset.more){ const eid=this._eid(target.dataset.more); if(!eid) return; this.dispatchEvent(new CustomEvent("hass-more-info",{detail:{entityId:eid},bubbles:true,composed:true})); return; }
    if(target.dataset.toggle){
      const id=target.dataset.toggle, eid=this._eid(id); if(!eid) return; const domain=eid.split(".")[0], next=this._on(id)?"off":"on";
      this._setOptimistic(id,next); this._hass.callService(domain,"toggle",{entity_id:eid}); return;
    }
    if(target.dataset.coverToggle){
      const id=target.dataset.coverToggle, inverted=target.dataset.inverted==="1", open=this._coverOpen(id,inverted), next=open?"closed":"open";
      this._optimistic.set(id,{state:next,at:Date.now()}); this._renderPage(false);
      const eid=this._eid(id); if(!eid) return; this._hass.callService("cover",open?"close_cover":"open_cover",{entity_id:eid}); return;
    }
    if(target.dataset.climate){
      const id=target.dataset.climate, mode=target.dataset.mode; this._setOptimistic(id,mode);
      const eid=this._eid(id); if(!eid) return; this._hass.callService("climate","set_hvac_mode",{entity_id:eid,hvac_mode:mode}); return;
    }
    if(target.dataset.climateTemp){
      const id=target.dataset.climateTemp, current=Number(this._attr(id,"temperature",24)), delta=Number(target.dataset.delta||0);
      const eid=this._eid(id); if(!eid) return; this._hass.callService("climate","set_temperature",{entity_id:eid,temperature:current+delta}); return;
    }
    if(target.dataset.script){
      const id=target.dataset.script, btn=target.closest?.("button") || target;
      btn.classList.add("pressed");
      const oldText=btn.textContent; btn.textContent="FATTO";
      setTimeout(()=>{ btn.classList.remove("pressed"); btn.textContent=oldText; },1200);
      const eid=this._eid(id); if(!eid) return; this._hass.callService("script","turn_on",{entity_id:eid}); return;
    }
    if(target.dataset.vacuum){ const eid=this._eid("vacuum.robot_vacuum"); if(!eid) return; this._hass.callService("vacuum",target.dataset.vacuum,{entity_id:eid}); return; }
    if(target.dataset.alarm){ const eid=this._eid("alarm_control_panel.home_alarm"); if(!eid) return; this._hass.callService("alarm_control_panel",target.dataset.alarm,{entity_id:eid}); return; }
    if(target.dataset.humidityDelta){
      const id="humidifier.garage_dehumidifier", current=Number(this._attr(id,"humidity",this._attr(id,"target_humidity",55))), delta=Number(target.dataset.humidityDelta||0);
      const eid=this._eid(id); if(!eid) return; this._hass.callService("humidifier","set_humidity",{entity_id:eid,humidity:Math.max(30,Math.min(80,current+delta))}); return;
    }
  }
