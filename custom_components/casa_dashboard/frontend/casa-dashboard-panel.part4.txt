his._on("binary_sensor.ev_charging_at_home")) return {level:"blue",text:"EV in carica"};
        if(this._on("binary_sensor.garage_dehumidifier_tank_full")) return {level:"bad",text:"Serbatoio pieno"};
        return {level:"good",text:"Tutto ok"}; }
    }
    return {level:"good",text:"Tutto ok"};
  }

  _renderNav(force=false) {
    const nav=this.shadowRoot.querySelector("#nav"); if(!nav) return;

    } else {
      nav.querySelectorAll(".nav-btn").forEach(btn=>btn.classList.toggle("active",btn.dataset.nav===this._active));
    }
  }
