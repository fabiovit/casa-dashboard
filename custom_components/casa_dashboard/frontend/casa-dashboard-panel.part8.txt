  _roomCard(room) {

    if(room==="cucina") tags=[`🌡️ ${this._fmt("sensor.kitchen_temperature",1,"°C")}`,`💧 ${this._fmt("sensor.kitchen_humidity",0,"%")}`];
    if(room==="camera") tags=[`🌡️ ${this._fmt("sensor.bedroom_temperature",1,"°C")}`,`💧 ${this._fmt("sensor.bedroom_humidity",0,"%")}`];
    if(room==="balcone") tags=[`🌡️ ${this._fmt("sensor.weather_outdoor_temperature",1,"°C")}`,`💨 ${this._fmt("sensor.weather_wind_gust",0,"km/h")}`];

      <div class="room-heading">
        <div class="room-orb ${s.level}"><ha-icon icon="${m.icon}"></ha-icon></div>
        <div class="room-arrow"><ha-icon icon="mdi:arrow-top-right"></ha-icon></div>
      </div>
      <div class="room-copy">
        <div class="room-eyebrow">${m.subtitle}</div>
        <h3>${m.title}</h3>
        <div class="room-status-line ${s.level}">${this._esc(s.text)}</div>
